   170→function traceNormalizeJsonText(text) {
   171→  return String(text || "")
   172→    .replace(/[“”]/g, "\"")
   173→    .replace(/[‘’]/g, "'");

   176→function traceFindJsonEnd(text, start) {
   177→  let depth = 0;
   178→  let inString = false;
   179→  let escaped = false;
   180→  for (let i = start; i < text.length; i += 1) {
   181→    const ch = text[i];
   182→    if (inString) {
   183→      if (escaped) {
   184→        escaped = false;
   185→      } else if (ch === "\\") {
   186→        escaped = true;
   187→      } else if (ch === "\"") {
   188→        inString = false;
   189→      }
   190→      continue;
   191→    }
   192→    if (ch === "\"") {
   193→      inString = true;
   194→    } else if (ch === "{") {
   195→      depth += 1;
   196→    } else if (ch === "}") {
   197→      depth -= 1;
   198→      if (depth === 0) return i + 1;
   199→    }
   200→  }
   201→  return -1;
   202→}

   204→function traceExtractMetadataObject(text) {
   205→  const normalized = traceNormalizeJsonText(text);
   206→  const first = normalized.indexOf("{");
   207→  if (first < 0) throw new Error("Metadata 页面里没有找到 JSON 开头。");
   208→  const end = traceFindJsonEnd(normalized, first);
   209→  if (end < 0) throw new Error("Metadata JSON 没有完整结束。");
   210→  const rawJson = normalized.slice(first, end);
   211→  const job = JSON.parse(rawJson);
   212→  if (!job.uid) {
   213→    const uidMatch = normalized.match(/\bJ\d+\b/i);
   214→    if (uidMatch) job.uid = uidMatch[0].toUpperCase();
   215→  }
   216→  if (!Number.isInteger(job.uid_num)) {
   217→    job.uid_num = traceJobNum(job.uid);
   218→  }
   219→  return { job, rawJson };
   220→}

   456→function traceCollectOverviewSelect2DImage() {
   457→  const candidates = [];
   458→  const seen = new Set();
   459→  function pushCandidate(item) {
   460→    if (!item || !item.url || seen.has(item.url)) return;
   461→    seen.add(item.url);
   462→    candidates.push(item);
   463→  }
   464→  const pngLinks = Array.from(document.querySelectorAll("a[href*='/api/log_image/']"))
   465→    .filter(traceIsPngLogLink);
   466→  for (const link of pngLinks) {
   467→    const text = traceNearestSelectedClassText(link);
   468→    const match = text.match(/Selected\s+(\d+)\s+classes/i);
   469→    if (!match) continue;
   470→    pushCandidate({
   471→      url: traceAbsoluteUrl(link.getAttribute("href") || ""),
   472→      text,
   473→      classes_selected: Number(match[1]),
   474→      timestamp: traceSelectedClassesTimestamp(text),
   475→      index: candidates.length,
   476→      source: "png_link_direct"
   477→    });
   478→  }
   479→  for (const container of traceLogContainerCandidates()) {
   480→    const match = container.text.match(/Selected\s+(\d+)\s+classes/i);
   481→    const pngLink = traceFirstPngLikeLogLink(container.el);
   482→    const img = container.el.querySelector("img[src*='/api/log_image/']");
   483→    const rawUrl = (pngLink && pngLink.getAttribute("href")) || (img && img.getAttribute("src")) || "";
   484→    if (!rawUrl || /download_result_file|\.pdf(?:$|\?)/i.test(rawUrl)) continue;
   485→    pushCandidate({
   486→      url: traceAbsoluteUrl(rawUrl),
   487→      text: container.text,
   488→      classes_selected: Number(match[1]),
   489→      timestamp: traceSelectedClassesTimestamp(container.text),
   490→      index: container.index,
   491→      source: pngLink ? "png_link" : "inline_image"
   492→    });
   493→  }
   494→  const elements = Array.from(document.querySelectorAll("a[href*='/api/log_image/'], img[src*='/api/log_image/']"));
   495→  for (const el of elements) {
   496→    const rawUrl = el.getAttribute("href") || el.getAttribute("src") || "";
   497→    if (!rawUrl || /download_result_file|\.pdf(?:$|\?)/i.test(rawUrl)) continue;
   498→    const text = traceElementLogText(el);
   499→    const label = (el.innerText || el.textContent || el.getAttribute("alt") || "").trim().toLowerCase();
   500→    if (label === "pdf" || /\bpdf\b/i.test(label)) continue;
   501→    const match = text.match(/Selected\s+(\d+)\s+classes/i);
   502→    if (!match) continue;
   503→    pushCandidate({
   504→      url: traceAbsoluteUrl(rawUrl),
   505→      text,
   506→      classes_selected: Number(match[1]),
   507→      timestamp: traceSelectedClassesTimestamp(text),
   508→      index: candidates.length,
   509→      source: el.tagName === "A" ? "png_link_fallback" : "image_fallback"
   510→    });
   511→  }
   512→  return traceBestSelectImageCandidate(candidates);
   513→}

   515→function traceScrollableElements() {

   517→  for (const element of document.querySelectorAll("*")) {

   555→    const item = traceCollectOverviewSelect2DImage();
   556→    if (item && item.url && !seen.has(item.url)) {

   586→        const currentTop = traceGetScrollTop(scroller);
   587→        const nextTop = start === 0
   588→          ? Math.min(maxTop, currentTop + amount)
   589→          : Math.max(0, currentTop - amount);

   597→        previousTop = afterTop;
   598→        if (stalled >= 6) break;
   599→      }

   601→  }
   602→
   603→  return traceBestSelectImageCandidate(candidates);

   920→function traceFocusedMermaid(nodes, edges, startUid) {

   937→    if (!merged.has(key)) merged.set(key, { source: edge.source, target: edge.target, kinds: new Set() });

   942→  const order = ["particle", "volume", "mask", "template", "exposure", "parent"];
   943→  for (const item of merged.values()) {
   944→    const label = order.filter((kind) => item.kinds.has(kind)).map((kind) => labels[kind]).join(" + ");
   945→    lines.push(`  ${item.source} -- "${label || "source"}" --> ${item.target}`);
   946→  }
   947→
   948→  lines.push("  classDef final fill:#fee2e2,stroke:#b91c1c,stroke-width:3px,color:#111827;");
   949→  lines.push("  classDef major fill:#dbeafe,stroke:#1d4ed8,stroke-width:2px,color:#111827;");

   952→}
