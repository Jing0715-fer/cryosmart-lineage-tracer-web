    55→/** HTML-escape a string for safe interpolation into HTML. */
    56→export function htmlEscape(value: unknown): string {
    57→  return String(value ?? "")
    58→    .replace(/&/g, "&amp;")
    59→    .replace(/</g, "&lt;")
    60→    .replace(/>/g, "&gt;")
    61→    .replace(/"/g, "&quot;")
    62→    .replace(/'/g, "&#39;");
    63→}

    95→/** Infer the lineage kind ("particle" / "volume" / ...) of an edge. */
    96→// duplicated from lineage.ts to avoid circular import
    97→export function edgeKind(edge: LineageEdge): string {
    98→  if (["particle", "volume", "mask", "template", "exposure"].includes(edge.input_type)) {
    99→    return edge.input_type;
   100→  }
   101→  const types = (edge.slots || []).map((slot) => slot.result_type || "").join(" ");
   102→  for (const kind of ["particle", "volume", "mask", "template", "exposure"]) {
   103→    if (types.includes(kind)) return kind;
   104→  }
   105→  return edge.input_type || "parent";
   106→}

   111→  if (edge.kind) return edge.kind;

   113→  if (Array.isArray(edge.kinds) && edge.kinds.length) return edge.kinds[0];
   114→  return "parent";

   189→function pixelSizeNumber(value: unknown): number | null {
   190→  const number = Number(value);
   191→  return Number.isFinite(number) && number > 0 && number < 100
   192→    ? Math.round(number * 10000) / 10000
   193→    : null;
   194→}
   195→
   196→function formatPixelSize(value: unknown): string {
   197→  const number = pixelSizeNumber(value);
   198→  if (!number) return "";
   199→  return Number.isInteger(number)
   200→    ? String(number)
   201→    : number.toFixed(4).replace(/0+$/, "").replace(/\.$/, "");
   202→}

   205→// duplicated from lineage.ts to avoid circular import
   206→export function pixelSizeText(node: LineageNode | null | undefined): string {
   207→  const text = formatPixelSize(node && node.pixel_size_A);
   208→  return text ? `${text} Å/px` : "";
   209→}

   211→function resolutionNumber(value: unknown): number | null {
   212→  const number = Number(value);
   213→  return Number.isFinite(number) && number >= 1 && number <= 20
   214→    ? Math.round(number * 100) / 100
   215→    : null;
   216→}

   218→function formatBinFactor(value: number): string {
   219→  if (!Number.isFinite(value)) return "";
   220→  const rounded = Math.round(value);
   221→  return Math.abs(value - rounded) < 0.01
   222→    ? String(rounded)
   223→    : value.toFixed(2).replace(/\.?0+$/, "");
   224→}
   225→
   226→/** "3.2 Å" or empty string. */
   227→// duplicated from lineage.ts to avoid circular import
   228→export function resolutionText(node: LineageNode | null | undefined): string {
   229→  const value = node && resolutionNumber(node.resolution_A);
   230→  return value ? `${formatBinFactor(value)} Å` : "";
   231→}

   355→// duplicated from lineage.ts to avoid circular import
   356→export function reportHasRepickSeed(
   357→  uid: string,
   358→  state: LineageReportState,
   359→  visited: Set<string> = new Set(),
   360→  depth = 0,
   361→): boolean {
   362→  if (!uid || visited.has(uid) || depth > 8) return false;
   363→  if (state.repickSeedMemo && state.repickSeedMemo.has(uid)) return state.repickSeedMemo.get(uid)!;
   364→  visited.add(uid);
   365→  const node = state.nodeMap.get(uid);
   366→  if (!node) return false;
   367→  const finish = (value: boolean): boolean => {
   368→    if (state.repickSeedMemo) state.repickSeedMemo.set(uid, value);
   369→    return value;
   370→  };
   371→  if (reportIsVolumeSourceNode(node)) return finish(true);
   372→  if (reportIsRepickParticleProducer(node)) return finish(false);
   373→
   374→  const incoming = state.incomingByTarget.get(uid) || [];
   375→  for (const edge of incoming) {
   376→    const source = state.nodeMap.get(edge.source);
   377→    if (edge.family === "volume" || edge.kind === "mask") return finish(true);
   378→    if (edge.family === "particle" && reportIsVolumeSourceNode(source)) return finish(true);
   379→    if (
   380→      reportIsRepickSetupNode(node) ||
   381→      reportIsRepickSetupNode(source) ||
   382→      edge.family === "particle"
   383→    ) {
   384→      if (reportHasRepickSeed(edge.source, state, new Set(visited), depth + 1)) return finish(true);
   385→    }
   386→  }
   387→  return finish(false);
   388→}

   390→// duplicated from lineage.ts to avoid circular import
   391→export function reportFeedsVolumeMainline(
   392→  uid: string,
   393→  state: LineageReportState,
   394→  visited: Set<string> = new Set(),
   395→  depth = 0,
   396→): boolean {
   397→  if (!uid || visited.has(uid) || depth > 10) return false;
   398→  visited.add(uid);
   399→  const node = state.nodeMap.get(uid);
   400→  if (!node) return false;
   401→  if (depth > 0 && reportIsVolumeSourceNode(node)) return true;
   402→  const outgoing = state.outgoingBySource ? state.outgoingBySource.get(uid) || [] : [];
   403→  for (const edge of outgoing) {
   404→    const target = state.nodeMap.get(edge.target);
   405→    if (!target) continue;
   406→    if (
   407→      edge.family === "particle" ||
   408→      edge.family === "volume" ||
   409→      edge.family === "template" ||
   410→      /model/i.test(edge.kind || "")
   411→    ) {
   412→      if (reportFeedsVolumeMainline(edge.target, state, new Set(visited), depth + 1)) return true;
   413→    }
   414→  }
   415→  return false;
   416→}

   418→// duplicated from lineage.ts to avoid circular import
   419→function reportRepickSeedSourceRounds(
   420→  incoming: NormalizedLineageEdge[],
   421→  state: LineageReportState,
   422→  visited: Set<string> = new Set(),
   423→): number[] {
   424→  return incoming
   425→    .map((edge) => {
   426→      const sourceNode = state.nodeMap.get(edge.source);
   427→      const directSeed =
   428→        edge.family === "volume" ||
   429→        edge.kind === "mask" ||
   430→        (edge.family === "particle" && reportIsVolumeSourceNode(sourceNode));
   431→      const inheritedSeed = reportIsRepickSetupNode(sourceNode) && reportHasRepickSeed(edge.source, state);
   432→      if (!directSeed && !inheritedSeed) return null;
   433→      return reportLineageRound(edge.source, state, new Set(visited));
   434→    })
   435→    .filter((value): value is number => Number.isInteger(value));
   436→}

   459→/**
   460→ * The "round" of a job — 0 for imports, 1 for first picking round, 2 for a
   461→ * repick after volume mainline, etc.
   462→ *
   463→ * `reportLineageRound` is the single most important function in the
   464→ * report layer: it dictates which round column a node appears in on the
   465→ * picture-flow diagram and the outline. Memoised via `state.roundMemo`.
   466→ */
   467→// duplicated from lineage.ts to avoid circular import
   468→export function reportLineageRound(
   469→  uid: string,
   470→  state: LineageReportState,
   471→  visited: Set<string> = new Set(),
   472→): number {
   473→  if (!uid || visited.has(uid)) return 0;
   474→  if (state.roundMemo && state.roundMemo.has(uid)) return state.roundMemo.get(uid)!;
   475→  visited.add(uid);
   476→  const node = state.nodeMap.get(uid);
   477→  if (!node) return 0;
   478→  const type = node.job_type || "";
   479→  const finish = (value: number): number => {
   480→    if (state.roundMemo) state.roundMemo.set(uid, value);
   481→    return value;
   482→  };
   483→
   484→  if (/import_(movies|micrographs)/i.test(type)) return finish(0);
   485→  if (/import_particles/i.test(type)) return finish(1);
   486→
   487→  const incoming = state.incomingByTarget.get(uid) || [];
   488→  const maxSourceRound = reportMaxRoundFromEdges(incoming, state, visited);
   489→  const particleSourceRound = reportParticleSourceRound(incoming, state, visited);
   490→  const seedSourceRounds = reportRepickSeedSourceRounds(incoming, state, visited);
   491→  const seedRound = seedSourceRounds.length ? Math.max(...seedSourceRounds) : null;
   492→
   493→  if (reportIsRepickSetupNode(node)) {
   494→    if (seedRound !== null && reportFeedsVolumeMainline(uid, state)) {
   495→      return finish(Math.max(2, seedRound + 1));
   496→    }
   497→    return finish(Math.max(1, (particleSourceRound ?? maxSourceRound) || seedRound || 1));
   498→  }
   499→
   500→  if (reportIsRepickParticleProducer(node)) {
   501→    const setupSourceRounds = incoming
   502→      .filter((edge) => reportIsRepickSetupNode(state.nodeMap.get(edge.source)))
   503→      .map((edge) => reportLineageRound(edge.source, state, new Set(visited)));
   504→    if (setupSourceRounds.length) {
   505→      return finish(Math.max(1, ...setupSourceRounds));
   506→    }
   507→    if (seedRound !== null && reportFeedsVolumeMainline(uid, state)) {
   508→      return finish(Math.max(2, seedRound + 1));
   509→    }
   510→    return finish(Math.max(1, (particleSourceRound ?? maxSourceRound) || seedRound || 1));
   511→  }
   512→
   513→  if (reportIsPickingNode(node)) {
   514→    return finish(Math.max(1, (particleSourceRound ?? maxSourceRound) || 1));
   515→  }
   516→
   517→  if (reportIsParticleAuxNode(node)) {
   518→    return finish(particleSourceRound ?? maxSourceRound);
   519→  }
   520→
   521→  if (/class_2D|select_2D|rebalance_classes_2D|class_probability_filter/i.test(type)) {
   522→    return finish(Math.max(1, particleSourceRound ?? maxSourceRound));
   523→  }
   524→
   525→  if (

   643→/** "颗粒 12345 · 照片 678 · 3.2 Å" — one-line metric string for a node. */
   644→// duplicated from lineage.ts to avoid circular import
   645→export function reportMetricText(node: LineageNode | null | undefined, compact = false): string {
   646→  if (!node) return "";
   647→  const parts: string[] = [];
   648→  if (node.micrograph_count !== null && node.micrograph_count !== undefined) {
   649→    parts.push(`照片 ${fmt(node.micrograph_count)}`);
   650→  }
   651→  if (pixelSizeText(node)) parts.push(`pixel ${pixelSizeText(node)}`);
   652→  if (node.particle_count !== null && node.particle_count !== undefined) {
   653→    parts.push(`颗粒 ${fmt(node.particle_count)}`);
   654→  }
   655→  if (node.class_count !== null && node.class_count !== undefined) {
   656→    parts.push(`class ${fmt(node.class_count)}`);
   657→  }
   658→  if (node.volume_count !== null && node.volume_count !== undefined) {
   659→    parts.push(`volume ${fmt(node.volume_count)}`);
   660→  }
   661→  const res = resolutionText(node);
   662→  if (res) parts.push(res);
   663→  const bin = extractionBinText(node);
   664→  if (bin) parts.push(bin);
   665→  // popup.js branches on `compact` but returns the same string either way;
   666→  // keep the branch for verbatim parity.
   667→  return compact ? parts.join(" · ") : parts.join(" · ");
   668→}

   783→/**
   784→ * Walk back from `sourceUid` through same-family edges until we hit a node
   785→ * that is in `visible`. Memoised on the `incomingByTarget` map itself
   786→ * (`__traceVisibleMemo`).
   787→ */
   788→// duplicated from lineage.ts to avoid circular import
   789→export function reportTraceVisibleSources(
   790→  sourceUid: string,
   791→  family: EdgeFamily | string,
   792→  visible: Set<string>,
   793→  incomingByTarget: IncomingByTargetMap,
   794→  visited: Set<string> = new Set(),
   795→  depth = 0,
   796→): string[] {
   797→  const memo = incomingByTarget.__traceVisibleMemo;
   798→  const memoKey = `${sourceUid}\t${family}`;
   799→  const useMemo = visited.size === 0 && memo;
   800→  if (useMemo && memo!.has(memoKey)) return memo!.get(memoKey)!;
   801→  if (!sourceUid || visited.has(sourceUid) || depth > 8) return [];
   802→  visited.add(sourceUid);
   803→  if (visible.has(sourceUid)) {
   804→    const result: string[] = [sourceUid];
   805→    if (useMemo) memo!.set(memoKey, result);
   806→    return result;
   807→  }
   808→  const allIncoming = incomingByTarget.get(sourceUid) || [];
   809→  let incoming = allIncoming.filter((edge) => edge.family === family);
   810→  if (!incoming.length) incoming = allIncoming;
   811→  if (!incoming.length) return [];
   812→  const results: string[] = [];
   813→  for (const edge of incoming) {
   814→    results.push(
   815→      ...reportTraceVisibleSources(edge.source, family, visible, incomingByTarget, new Set(visited), depth + 1),
   816→    );
   817→  }
   818→  const result = Array.from(new Set(results)).sort((a, b) => reportJobNum(a) - reportJobNum(b));
   819→  if (useMemo) memo!.set(memoKey, result);
   820→  return result;
   821→}

   864→/** Visible ancestor uids (one per family) for an outline mini-node. */
   865→// duplicated from lineage.ts to avoid circular import
   866→export function reportOutlineRefs(
   867→  uid: string,
   868→  state: LineageReportState,
   869→): Array<[string, EdgeFamily | string]> {
   870→  const refs = new Map<string, [string, EdgeFamily | string]>();
   871→  for (const edge of state.incomingByTarget.get(uid) || []) {
   872→    for (const source of reportTraceVisibleSources(
   873→      edge.source,
   874→      edge.family,
   875→      state.visible,
   876→      state.incomingByTarget,
   877→    )) {
   878→      if (!state.visible.has(source) || source === uid) continue;
   879→      const key = `${source}\t${edge.family}`;
   880→      if (!refs.has(key)) refs.set(key, [source, edge.family]);
   881→    }
   882→  }
   883→  const familyOrder: Record<string, number> = {
   884→    exposure: 1,
   885→    micrograph: 1,
   886→    particle: 2,
   887→    volume: 3,
   888→    template: 4,
   889→    other: 5,
   890→  };
   891→  return Array.from(refs.values()).sort((a, b) => {
   892→    const byJob = reportJobNum(a[0]) - reportJobNum(b[0]);
   893→    if (byJob) return byJob;
   894→    return (familyOrder[a[1] as string] || 9) - (familyOrder[b[1] as string] || 9);
   895→  });
   896→}

   901→  targetUid: string,
   902→  sourceUid: string,
   903→  family: EdgeFamily | string,
   904→  state: LineageReportState,
   905→): string {
   906→  if (state.visible.has(sourceUid)) return "";
   907→  const refs: string[] = [];
   908→  for (const edge of state.incomingByTarget.get(sourceUid) || []) {
   909→    if (edge.family === family) {
   910→      refs.push(
   911→        ...reportTraceVisibleSources(edge.source, family, state.visible, state.incomingByTarget),
   912→      );
   913→    }
   914→  }
   915→  if (!refs.length) {

   919→      );

   921→  }
   922→  const dedupedRefs = Array.from(new Set(refs))
   923→    .filter((uid) => uid !== sourceUid && state.visible.has(uid))
   924→    .sort((a, b) => reportJobNum(a) - reportJobNum(b));
   925→  if (!dedupedRefs.length) return "";

   934→        metric ? ` ${escHtml(metric)}` : ""
   935→      }</div>`;
   936→    })
   937→    .join("");
   938→  return `<div class="up-route">${route}</div><div class="up-list">${lines}</div>`;
   939→}

  1215→/** The left outline panel — stages → phases → mini-node grid. */
  1216→export function reportOutline(summary: LineageSummary, state: LineageReportState): string {
  1217→  const byStage = new Map<string, LineageNode[]>();
  1218→  for (const node of state.outlineNodes) {
  1219→    const stage = reportStageName(node, summary, state);
  1220→    if (!byStage.has(stage)) byStage.set(stage, []);
  1221→    byStage.get(stage)!.push(node);
  1222→  }
  1223→  const roundStages = Array.from(byStage.keys())
  1224→    .filter((stage) => /^Round \d+$/i.test(stage))
  1225→    .sort((a, b) => Number(a.match(/\d+/)?.[0] || 0) - Number(b.match(/\d+/)?.[0] || 0));
  1226→  const stageOrder = ["Micrographs", ...roundStages, "Auxiliary"];
  1227→  const phaseOrder = [
  1228→    "导入 / 预处理",
  1229→    "挑颗粒 / 提取",
  1230→    "2D",
  1231→    "再挑颗粒 / 提取",
  1232→    "初始建模",
  1233→    "refine / final",
  1234→    "附属",
  1235→  ];
  1236→  return stageOrder
  1237→    .filter((stage) => (byStage.get(stage) || []).length)
  1238→    .map((stage, idx, stages) => {
  1239→      const nodes = byStage.get(stage) || [];
  1240→      const byPhase = new Map<string, LineageNode[]>();
  1241→      for (const node of nodes) {
  1242→        const phase = reportPhaseName(node, summary, state);
  1243→        if (!byPhase.has(phase)) byPhase.set(phase, []);
  1244→        byPhase.get(phase)!.push(node);
  1245→      }
  1246→      const phaseHtml = phaseOrder
  1247→        .filter((phase) => (byPhase.get(phase) || []).length)
  1248→        .map((phase) => {
  1249→          const phaseNodes = byPhase
  1250→            .get(phase)!
  1251→            .map((node) => reportMiniNode(node, state))
  1252→            .join("");
  1253→          return `<div class="phase"><div class="phase-label">${escHtml(
  1254→            phase,
  1255→          )}</div><div class="stage-grid">${phaseNodes}</div></div>`;
  1256→        })
  1257→        .join("");
  1258→      const arrow = idx < stages.length - 1 ? `<div class="stage-arrow">&darr;</div>` : "";
  1259→      return `<div class="stage"><h3>${escHtml(stage)}</h3>${phaseHtml}</div>${arrow}`;
  1260→    })
  1261→    .join("");
  1262→}

  1746→  state: LineageReportState,

  1780→    round > 1 ? " repicking" : ""
