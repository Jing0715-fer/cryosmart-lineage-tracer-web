    72→    72→def parse_corr(text):
    73→    73→    if text is None:
    74→    74→        return None
    75→    75→    if isinstance(text, (int, float)):
    76→    76→        value = float(text)
    77→    77→        return value if -1.0 <= value <= 1.0 else None
    78→    78→    text = str(text)
    79→    79→    m = re.search(r"correlation\s*=?\s*([-+]?\d*\.?\d+)", text, re.I)
    80→    80→    if m:
    81→    81→        value = float(m.group(1))
    82→    82→        return value if -1.0 <= value <= 1.0 else None
    83→    83→    return None

    85→    85→def run_capture(session, cmd, command_log=None):

    91→    91→        cxrun(session, "log clear")

    93→    93→        cxrun(session, "log save " + q(log_path))

    95→    95→            log_text = f.read()
    96→    96→        return result, log_text

    99→    99→            os.unlink(log_path)

   273→   273→def target_map_groups_for_edge(edge, target_node, item_by_key):
   274→   274→    target_uid = edge.get("target")

   279→   279→        for map_item in normal_map_assets(target_node):
   280→   280→            group = map_group_key(map_item.get("group") or "volume")

   286→   286→        candidates.append(source_group)

   288→   288→        for group in target_groups:

   290→   290→                candidates.append(group)

   292→   292→        candidates.append("volume")

   294→   294→        candidates.append(target_groups[0])

   297→   297→def build_map_lineage(summary, items):
   298→   298→    item_by_key = {item["key"]: item for item in items}
   299→   299→    nodes_by_uid = {node.get("uid"): node for node in summary.get("nodes") or [] if node.get("uid")}
   300→   300→    edges = []
   301→   301→    seen = set()
   302→   302→    warnings = []
   303→   303→
   304→   304→    for edge in summary.get("edges") or []:
   305→   305→        if edge.get("input_type") != "volume":
   306→   306→            continue
   307→   307→        source_uid = edge.get("source")
   308→   308→        target_uid = edge.get("target")
   309→   309→        source_group = map_group_key(edge.get("source_group"))
   310→   310→        source_key = canonical_map_key(source_uid, source_group)
   311→   311→        if source_key not in item_by_key:
   312→   312→            warnings.append(f"skip volume edge {source_uid}.{source_group} -> {target_uid}: source MRC missing")
   313→   313→            continue
   314→   314→        target_node = nodes_by_uid.get(target_uid)
   315→   315→        for target_group in target_map_groups_for_edge(edge, target_node, item_by_key):
   316→   316→            target_key = canonical_map_key(target_uid, target_group)
   317→   317→            edge_key = (source_key, target_key)
   318→   318→            if edge_key in seen:
   319→   319→                continue
   320→   320→            seen.add(edge_key)
   321→   321→            edges.append({
   322→   322→                "source": source_key,
   323→   323→                "target": target_key,
   324→   324→                "reason": f"{source_uid}.{source_group} -> {target_uid}.{target_group}",
   325→   325→            })
   326→   326→
   327→   327→    incoming = {}
   328→   328→    outgoing = {}
   329→   329→    for edge in edges:
   330→   330→        outgoing.setdefault(edge["source"], []).append(edge["target"])
   331→   331→        incoming.setdefault(edge["target"], []).append(edge["source"])
   332→   332→
   333→   333→    start_keys = [key for key in item_by_key if key not in incoming]
   334→   334→    if not start_keys:
   335→   335→        start_keys = sorted(item_by_key)
   336→   336→        warnings.append("No start map inferred; treating every downloaded map as a possible start.")
   337→   337→
   338→   338→    chains = []
   339→   339→
   340→   340→    def walk(start_key, current_key, path):
   341→   341→        children = sorted(outgoing.get(current_key) or [])
   342→   342→        if not children:
   343→   343→            chains.append({"start": start_key, "terminal": current_key, "keys": list(path)})
   344→   344→            return
   345→   345→        for child in children:
   346→   346→            if child in path:
   347→   347→                warnings.append(f"Cycle detected in map lineage: {' -> '.join(path + [child])}")
   348→   348→                continue
   349→   349→            walk(start_key, child, path + [child])
   350→   350→
   351→   351→    for start_key in sorted(start_keys, key=lambda key: (parse_job_num(key), key)):
   352→   352→        walk(start_key, start_key, [start_key])
   353→   353→    if not chains:
   354→   354→        chains = [{"start": key, "terminal": key, "keys": [key]} for key in sorted(item_by_key)]
   355→   355→
   356→   356→    return {
   357→   357→        "items": item_by_key,
   358→   358→        "edges": edges,
   359→   359→        "incoming": incoming,
   360→   360→        "outgoing": outgoing,
   361→   361→        "starts": sorted(start_keys, key=lambda key: (parse_job_num(key), key)),
   362→   362→        "chains": chains,
   363→   363→        "warnings": warnings,
   364→   364→    }

   542→   542→# ==================== 视角自动优化函数（来自第三个脚本） ====================
   543→   543→def score_current_view(session):
   544→   544→    """截取当前视角缩略图，计算非白色像素比例作为评分（地图覆盖率）"""
   545→   545→    if np is None or Image is None:
   546→   546→        return 0.0
   547→   547→    with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
   548→   548→        path = tmp.name
   549→   549→    try:
   550→   550→        cxrun(session, f"save {q(path)} width {SCORE_THUMB_SIZE} height {SCORE_THUMB_SIZE} supersample 1 transparentBackground false")
   551→   551→        img = Image.open(path).convert("RGB")
   552→   552→        arr = np.array(img)
   553→   553→        bg = np.all(arr > 240, axis=-1)
   554→   554→        score = float((arr.shape[0] * arr.shape[1] - np.sum(bg)) / (arr.shape[0] * arr.shape[1]))
   555→   555→        return score
   556→   556→    except Exception as e:
   557→   557→        session.logger.warning(f"视角评分失败: {e}")
   558→   558→        return 0.0
   559→   559→    finally:
   560→   560→        if os.path.exists(path):
   561→   561→            os.unlink(path)

   563→   563→def select_best_90_view(session, test_vol):
   564→   564→    """
   565→   565→    基于当前视角（需先手动调好基准），测试5个方向（基准、Y±90、X±90），
   566→   566→    选择地图占画面比例最大的方向，并应用该视角。
   567→   567→    """
   568→   568→    session.logger.info("===== 90° 视角自动优化 =====")
   569→   569→
   570→   570→    # 保存当前视角为基准
   571→   571→    cxrun(session, "view name base_view")
   572→   572→    cxrun(session, "wait 1")
   573→   573→
   574→   574→    # 只显示测试地图
   575→   575→    all_vols = volume_models(session)
   576→   576→    for vol in all_vols:
   577→   577→        if vol.id_string == test_vol.id_string:
   578→   578→            cxrun(session, f"show #{vol.id_string} models")
   579→   579→        else:
   580→   580→            cxrun(session, f"hide #{vol.id_string} models")
   581→   581→    cxrun(session, "wait 1")
   582→   582→
   583→   583→    candidates = [
   584→   584→        ("base", 0, 0),
   585→   585→        ("y+90", 0, 90),
   586→   586→        ("y-90", 0, -90),
   587→   587→        ("x+90", 90, 0),
   588→   588→        ("x-90", -90, 0),
   589→   589→    ]
   590→   590→
   591→   591→    best_score = -1.0
   592→   592→    best_desc = "base"
   593→   593→    best_rot = (0, 0)
   594→   594→
   595→   595→    for desc, x_rot, y_rot in candidates:
   596→   596→        cxrun(session, "view base_view")
   597→   597→        cxrun(session, "wait 1")
   598→   598→        if x_rot != 0:
   599→   599→            cxrun(session, f"turn x {x_rot}")
   600→   600→        if y_rot != 0:
   601→   601→            cxrun(session, f"turn y {y_rot}")
   602→   602→        cxrun(session, "wait 1")
   603→   603→        score = score_current_view(session)
   604→   604→        session.logger.info(f"  {desc}: X={x_rot}°, Y={y_rot}°, Score={score:.4f}")
   605→   605→        if score > best_score:
   606→   606→            best_score = score
   607→   607→            best_desc = desc
   608→   608→            best_rot = (x_rot, y_rot)
   609→   609→
   610→   610→    # 应用最佳视角
   611→   611→    session.logger.info(f"最佳视角: {best_desc} (X={best_rot[0]}°, Y={best_rot[1]}°), Score={best_score:.4f}")
   612→   612→    cxrun(session, "view base_view")
   613→   613→    cxrun(session, "wait 1")
   614→   614→    if best_rot[0] != 0:
   615→   615→        cxrun(session, f"turn x {best_rot[0]}")
   616→   616→    if best_rot[1] != 0:
   617→   617→        cxrun(session, f"turn y {best_rot[1]}")
   618→   618→    cxrun(session, "wait 1")
   619→   619→
   620→   620→    # 恢复显示所有地图
   621→   621→    for vol in all_vols:
   622→   622→        cxrun(session, f"show #{vol.id_string} models")
   623→   623→    cxrun(session, "wait 1")
   624→   624→
   625→   625→    return {"desc": best_desc, "x": best_rot[0], "y": best_rot[1], "score": best_score}
