   303→function parseResolutionText(value) {
   304→  const text = String(value || "").replace(/\s+/g, " ");
   305→  const patterns = [
   306→    /(?:resolution|res|FSC[^:]{0,40})[^0-9]{0,30}([0-9]+(?:\.[0-9]+)?)\s*(?:Å|A\b|angstrom)/i,
   307→    /([0-9]+(?:\.[0-9]+)?)\s*(?:Å|A\b|angstrom)[^.;]{0,60}(?:resolution|FSC|res)/i
   308→  ];
   309→  for (const pattern of patterns) {
   310→    const match = text.match(pattern);
   311→    const number = match && resolutionNumber(match[1]);
   312→    if (number) return number;
   313→  }
   314→  return null;
   315→}

   317→function resolutionFromObject(value, keyPath = "", depth = 0, seen = new Set()) {
   318→  if (value === null || value === undefined || depth > 7) return null;
   319→  if (typeof value === "number") {
   320→    return /resolution|fsc|res[_-]?a/i.test(keyPath) && !/threshold|cutoff/i.test(keyPath)
   321→      ? resolutionNumber(value)
   322→      : null;
   323→  }
   324→  if (typeof value === "string") {
   325→    if (/resolution|FSC|angstrom|Å|\bA\b/i.test(value) || /resolution|fsc/i.test(keyPath)) {
   326→      return parseResolutionText(value) || (/resolution|fsc/i.test(keyPath) ? resolutionNumber(value) : null);
   327→    }
   328→    return null;
   329→  }
   330→  if (typeof value !== "object" || seen.has(value)) return null;
   331→  seen.add(value);
   332→  if (Array.isArray(value)) {
   333→    for (const item of value) {
   334→      const found = resolutionFromObject(item, keyPath, depth + 1, seen);
   335→      if (found) return found;
   336→    }
   337→    return null;
   338→  }
   339→  for (const [key, item] of Object.entries(value)) {
   340→    const path = keyPath ? `${keyPath}.${key}` : key;
   341→    const found = resolutionFromObject(item, path, depth + 1, seen);
   342→    if (found) return found;
   343→  }
   344→  return null;
   345→}

   351→  return resolutionNumber(job && job.radwn_final_A) ||
   352→    resolutionNumber(job && job.final_resolution_A) ||
   353→    resolutionNumber(job && job.resolution_A) ||
   354→    resolutionNumber(job && job.fsc_resolution_A) ||
   355→    resolutionNumber(job && job.gold_standard_fsc_resolution_A) ||
   356→    resolutionFromObject(job && job.overview_assets && job.overview_assets.resolution_A, "overview_assets.resolution_A") ||
   357→    resolutionFromObject(job && job.overview_assets, "overview_assets", 0) ||
   358→    resolutionFromObject(job && job.output_result_groups, "output_result_groups", 0);

   554→function connectionEdges(job) {
   555→  const edges = [];
   556→  for (const inputGroup of job.input_slot_groups || []) {
   557→    for (const connection of inputGroup.connections || []) {
   558→      if (!connection.job_uid) continue;
   559→      edges.push({
   560→        source: connection.job_uid,
   561→        target: job.uid,
   562→        input_type: inputGroup.type,
   563→        input_name: inputGroup.name,
   564→        input_title: inputGroup.title,
   565→        source_group: connection.group_name,
   566→        slots: (connection.slots || []).map((slot) => ({
   567→          slot_name: slot.slot_name,
   568→          source_group: slot.group_name,
   569→          result_name: slot.result_name,
   570→          result_type: slot.result_type,
   571→          version: slot.version
   572→        }))
   573→      });
   574→    }
   575→  }
   576→  return edges;
   577→}

   580→  return (job.parents || [])

   593→function collectUpstream(projectJobs, startUid) {
   594→  const seen = new Map();
   595→  const edges = [];
   596→  const queue = [startUid];
   597→
   598→  while (queue.length) {
   599→    const uid = queue.shift();
   600→    if (seen.has(uid)) continue;
   601→    const job = projectJobs.get(uid);
   602→    if (!job) continue;
   603→    seen.set(uid, job);
   604→
   605→    const explicit = connectionEdges(job);
   606→    const sources = new Set(explicit.map((edge) => edge.source));
   607→    const jobEdges = explicit.concat(fallbackParentEdges(job, sources));
   608→    edges.push(...jobEdges);
   609→
   610→    for (const edge of jobEdges) {
   611→      if (!seen.has(edge.source)) queue.push(edge.source);
   612→    }
   613→  }
   614→
   615→  return { nodes: seen, edges };
   616→}

   631→function classSplits(job, baseUrl) {
   632→  const type = job.job_type || "";
   633→  if (!type.includes("abinit") && !type.includes("hetero") && !type.includes("class_3D")) return [];
   634→
   635→  let total = null;
   636→  const classes = new Map();
   637→  const outputImages = job.output_group_images || {};
   638→
   639→  for (const group of outputGroups(job, "particle")) {
   640→    if (group.name === "particles_all_classes") {
   641→      total = group.num_items;
   642→      continue;
   643→    }
   644→    const idx = parseClassIndex(group.name);
   645→    if (idx === null) continue;
   646→    if (!classes.has(idx)) classes.set(idx, {});
   647→    classes.get(idx).particle_count = group.num_items;
   648→  }
   649→
   650→  for (const group of outputGroups(job, "volume")) {
   651→    const idx = parseClassIndex(group.name);
   652→    if (idx === null) continue;
   653→    if (!classes.has(idx)) classes.set(idx, {});
   654→    const entry = classes.get(idx);
   655→    entry.volume_group = group.name;
   656→    entry.mrc_preview_url = logImageUrl(baseUrl, outputImages[group.name]);
   657→    entry.mrc_preview_src = entry.mrc_preview_url;
   658→    entry.mrc_preview_original_url = resultPreviewImageUrl(baseUrl, job.project_uid, job.uid, group.name);
   659→    entry.maps = (group.contains || [])
   660→      .filter((item) => item.type === "volume.blob" && item.name === "map")
   661→      .map((item) => ({
   662→        result_name: item.name,
   663→        download_url: `${baseUrl.replace(/\/$/, "")}/api/log_image/download_result_file/${job.project_uid}/${job.uid}.${group.name}.${item.name}`
   664→      }));
   665→  }
   666→
   667→  return Array.from(classes.entries()).sort((a, b) => a[0] - b[0]).map(([idx, entry]) => ({
   668→    class_index: idx,
   669→    particle_count: Number.isInteger(entry.particle_count) ? entry.particle_count : null,
   670→    particle_percent: Number.isInteger(entry.particle_count) && Number.isInteger(total) && total
   671→      ? Math.round(entry.particle_count / total * 10000) / 100
   672→      : null,
   673→    total_particles: total,
   674→    volume_group: entry.volume_group || null,
   675→    mrc_preview_url: entry.mrc_preview_url || null,
   676→    mrc_preview_src: entry.mrc_preview_src || null,
   677→    mrc_preview_original_url: entry.mrc_preview_original_url || null,
   678→    maps: entry.maps || []
   679→  }));
   680→}

   720→function focusedMermaid(nodes, edges, startUid) {
   721→  const lines = ["flowchart LR"];
   722→  const sortedNodes = Array.from(nodes.values()).sort((a, b) => (a.uid_num || 0) - (b.uid_num || 0));
   723→
   724→  for (const job of sortedNodes) {
   725→    const cls = importance(job, startUid);
   726→    lines.push(cls === "final"
   727→      ? `  ${job.uid}[["${nodeLabel(job)}"]]`
   728→      : `  ${job.uid}["${nodeLabel(job)}"]`);
   729→    lines.push(`  class ${job.uid} ${cls};`);
   730→  }
   731→
   732→  const merged = new Map();
   733→  for (const edge of edges) {
   734→    if (!nodes.has(edge.source) || !nodes.has(edge.target)) continue;
   735→    const key = `${edge.source}->${edge.target}`;
   736→    if (!merged.has(key)) merged.set(key, { source: edge.source, target: edge.target, kinds: new Set() });
   737→    merged.get(key).kinds.add(edgeKind(edge));
   738→  }
   739→
   740→  for (const item of Array.from(merged.values())) {
   741→    lines.push(`  ${item.source} -- "${focusedEdgeLabel(item.kinds)}" --> ${item.target}`);
   742→  }
   743→
   744→  lines.push("  classDef final fill:#fee2e2,stroke:#b91c1c,stroke-width:3px,color:#111827;");
   745→  lines.push("  classDef major fill:#dbeafe,stroke:#1d4ed8,stroke-width:2px,color:#111827;");
   746→  lines.push("  classDef small fill:#f3f4f6,stroke:#9ca3af,stroke-width:1px,color:#4b5563,font-size:11px;");
   747→  return `${lines.join("\n")}\n`;
   748→}

   772→function buildSummary(data, projectId, startUid, baseUrl) {
   773→  const projectJobs = new Map(
   774→    data.filter((job) => job.project_uid === projectId && job.uid).map((job) => [job.uid, job])
   775→  );
   776→  if (!projectJobs.has(startUid)) {
   777→    const latest = Array.from(projectJobs.values())
   778→      .filter((job) => Number.isInteger(job.uid_num))
   779→      .sort((a, b) => a.uid_num - b.uid_num)
   780→      .slice(-20)
   781→      .map((job) => [job.uid_num, job.uid, job.job_type]);
   782→    throw new Error(`${projectId}/${startUid} 不在当前 metadata 中。项目内 jobs=${projectJobs.size}，最新=${JSON.stringify(latest.at(-1))}`);
   783→  }
   784→
   785→  const { nodes, edges } = collectUpstream(projectJobs, startUid);
   786→  const startJob = projectJobs.get(startUid);
   787→  const nodeList = Array.from(nodes.values())
   788→    .sort((a, b) => (a.uid_num || 0) - (b.uid_num || 0))
   789→    .map((job) => jobNode(job, baseUrl, projectId));
   790→  const classJobs = nodeList
   791→    .filter((node) => Array.isArray(node.classes) && node.classes.length)
   792→    .map((node) => ({ uid: node.uid, job_type: node.job_type, classes: node.classes }));
   793→  const importJobs = Array.from(nodes.values())
   794→    .filter((job) => (job.job_type || "").startsWith("import_") || !(job.parents || []).length)
   795→    .map((job) => jobNode(job, baseUrl, projectId));
   796→
   797→  return normalizeLineageSummary({
   798→    ok: true,
   799→    project_uid: projectId,
   800→    base_url: baseUrl,
   801→    start_uid: startUid,
   802→    start_job: jobNode(startJob, baseUrl, projectId),
   803→    final_particle_count: maxGroupNumItems(startJob, "particle"),
   804→    final_micrograph_count: maxGroupNumItems(startJob, "exposure"),
   805→    final_resolution_A: resolutionFromJob(startJob),
   806→    resolution_note: "需要从 CryoSmart metadata/log/FSC 结果补充；jobs metadata 通常没有最终分辨率。",
   807→    map_download_urls: mapDownloadUrls(baseUrl, projectId, startUid),
   808→    nodes: nodeList,
   809→    edges,
   810→    import_or_leaf_jobs: importJobs,
   811→    class_split_jobs: classJobs,
   812→    focused_mermaid: focusedMermaid(nodes, edges, startUid)
   813→  });
   814→}

   944→function mimeForExtension(ext) {
   945→  switch (String(ext || "").toLowerCase()) {
   946→    case "png": return "image/png";
   947→    case "pdf": return "application/pdf";
   948→    case "xml": return "application/xml";
   949→    case "txt": return "text/plain";
   950→    default: return "application/octet-stream";
   951→  }
   952→}

   954→function sniffExtension(bytes, mime = "", sourceUrl = "") {
   955→  const arr = bytes instanceof Uint8Array ? bytes : new Uint8Array(bytes || []);
   956→  const lowerMime = String(mime || "").toLowerCase();
   957→  const lowerUrl = String(sourceUrl || "").toLowerCase();
   958→  if (arr.length >= 8 && arr[0] === 0x89 && arr[1] === 0x50 && arr[2] === 0x4e && arr[3] === 0x47) return "png";
   959→  if (arr.length >= 4 && arr[0] === 0x25 && arr[1] === 0x50 && arr[2] === 0x44 && arr[3] === 0x46) return "pdf";
   960→  if (lowerMime.includes("png")) return "png";
   961→  if (lowerMime.includes("pdf")) return "pdf";
   962→  if (lowerMime.includes("xml")) return "xml";
   963→  if (lowerMime.startsWith("text/")) return "txt";
   964→  const head = new TextDecoder("utf-8", { fatal: false }).decode(arr.slice(0, 512)).trimStart();
   965→  if (head.startsWith("<")) return "xml";
   966→  const urlExt = lowerUrl.match(/\.(png|pdf|txt|xml)(?:[?#]|$)/);
   967→  if (urlExt) return urlExt[1];
   968→  if (head) return "txt";
   969→  return null;
   970→}

   972→function replaceFilenameExtension(filename, ext) {
   973→  const wanted = String(ext || "").replace(/^\./, "").toLowerCase();
   974→  if (!wanted) return filename;
   975→  const parts = String(filename || "download").split("/");
   976→  const base = parts.pop() || "download";
   977→  const nextBase = /\.[A-Za-z0-9]{1,6}$/.test(base)
   978→    ? base.replace(/\.[^.]+$/, `.${wanted}`)
   979→    : `${base}.${wanted}`;
   980→  parts.push(nextBase);
   981→  return parts.join("/");
   982→}

   985→  const bytes = new Uint8Array(await blob.arrayBuffer());
   986→  const ext = sniffExtension(bytes, blob.type, sourceUrl);

   989→  return {

  1057→async function downloadFetchedAsset(item) {
  1058→  if (!item || !item.url || !item.filename) return null;
  1059→  if (item.mode === "fetch-image") {
  1060→    const urls = dedupeUrls([...(item.urls || []), item.url]);
  1061→    for (const url of urls) {
  1062→      const image = await fetchPptImage({ key: item.filename, url });
  1063→      if (!image) continue;
  1064→      const blob = new Blob([image.bytes], { type: image.mime || "image/png" });
  1065→      downloadBlob(item.filename, blob);
  1066→      return item.filename;
  1067→    }
  1068→    throw new Error(`图片下载失败：${item.filename}`);
  1069→  }
  1070→
  1071→  const urls = dedupeUrls([...(item.urls || []), item.url]);
  1072→  for (const url of urls) {
  1073→    try {
  1074→      const response = await fetch(url, { credentials: "include" });
  1075→      if (!response.ok) throw new Error(`HTTP ${response.status}`);
  1076→      const blob = await response.blob();
  1077→      const corrected = item.mode === "fetch-file"
  1078→        ? await correctedDownloadBlob(item.filename, blob, url)
  1079→        : { filename: item.filename, blob };
  1080→      downloadBlob(corrected.filename, corrected.blob);
  1081→      return corrected.filename;
  1082→    } catch (err) {
  1083→      try {
  1084→        const asset = await fetchAssetFromCurrentPage(url);
  1085→        const parsed = bytesFromDataUri(asset.data_url);
  1086→        if (!parsed) throw new Error("bad data url");
  1087→        const blob = new Blob([parsed.bytes], { type: parsed.mime || asset.content_type || "application/octet-stream" });
  1088→        const corrected = item.mode === "fetch-file"
  1089→          ? await correctedDownloadBlob(item.filename, blob, url)
  1090→          : { filename: item.filename, blob };
  1091→        downloadBlob(corrected.filename, corrected.blob);
  1092→        return corrected.filename;
  1093→      } catch (innerErr) {
  1094→        // Try the next URL candidate.
  1095→      }
  1096→    }
  1097→  }
  1098→  throw new Error(`文件下载失败：${item.filename}`);
  1099→}

  1101→async function downloadRemoteBatch(downloads) {
  1102→  const clean = (downloads || []).filter((item) => item && item.url && item.filename);
  1103→  if (!clean.length) return { ok: true, count: 0 };
  1104→
  1105→  const fetched = clean.filter((item) => /^fetch/.test(item.mode || ""));
  1106→  const direct = clean.filter((item) => !/^fetch/.test(item.mode || ""));
  1107→  let fetchCount = 0;
  1108→  const errors = [];
  1109→  for (const item of fetched) {
  1110→    try {
  1111→      await downloadFetchedAsset(item);
  1112→      fetchCount += 1;
  1113→    } catch (err) {
  1114→      errors.push(`${item.filename}: ${err.message}`);
  1115→    }
  1116→  }
  1117→
  1118→  if (!direct.length) return { ok: true, count: fetchCount, errors };
  1119→
  1120→  try {
  1121→    const result = await new Promise((resolve, reject) => {
  1122→      chrome.runtime.sendMessage({ action: "downloadCryoSmartFiles", downloads: direct }, (response) => {
  1123→        const err = chrome.runtime.lastError;
  1124→        if (err) reject(new Error(err.message));
  1125→        else resolve(response);
  1126→      });
  1127→    });
  1128→    if (!result || !result.ok) throw new Error((result && result.error) || "后台下载队列失败");
  1129→    return { ...result, count: (result.count || 0) + fetchCount, errors };
  1130→  } catch (err) {
  1131→    let count = fetchCount;
  1132→    for (const item of direct) {
  1133→      try {
  1134→        await downloadRemoteDirect(item.url, item.filename);
  1135→        count += 1;
  1136→      } catch (innerErr) {
  1137→        errors.push(`${item.filename}: ${innerErr.message}`);
  1138→      }
  1139→    }
  1140→    return { ok: true, count, fallback: true, errors };
  1141→  }
  1142→}

  1556→function reportLineageRound(uid, state, visited = new Set()) {
  1557→  if (!uid || visited.has(uid)) return 0;
  1558→  if (state.roundMemo && state.roundMemo.has(uid)) return state.roundMemo.get(uid);
  1559→  visited.add(uid);
  1560→  const node = state.nodeMap.get(uid);
  1561→  if (!node) return 0;
  1562→  const type = node.job_type || "";
  1563→  const finish = (value) => {
  1564→    if (state.roundMemo) state.roundMemo.set(uid, value);
  1565→    return value;
  1566→  };
  1567→
  1568→  if (/import_(movies|micrographs)/i.test(type)) return finish(0);
  1569→  if (/import_particles/i.test(type)) return finish(1);
  1570→
  1571→  const incoming = state.incomingByTarget.get(uid) || [];
  1572→  const maxSourceRound = reportMaxRoundFromEdges(incoming, state, visited);
  1573→  const particleSourceRound = reportParticleSourceRound(incoming, state, visited);
  1574→  const seedSourceRounds = reportRepickSeedSourceRounds(incoming, state, visited);
  1575→  const seedRound = seedSourceRounds.length ? Math.max(...seedSourceRounds) : null;
  1576→
  1577→  if (reportIsRepickSetupNode(node)) {
  1578→    if (seedRound !== null && reportFeedsVolumeMainline(uid, state)) {
  1579→      return finish(Math.max(2, seedRound + 1));
  1580→    }
  1581→    return finish(Math.max(1, (particleSourceRound ?? maxSourceRound) || seedRound || 1));
  1582→  }
  1583→
  1584→  if (reportIsRepickParticleProducer(node)) {
  1585→    const setupSourceRounds = incoming
  1586→      .filter((edge) => reportIsRepickSetupNode(state.nodeMap.get(edge.source)))
  1587→      .map((edge) => reportLineageRound(edge.source, state, new Set(visited)));
  1588→    if (setupSourceRounds.length) {
  1589→      return finish(Math.max(1, ...setupSourceRounds));
  1590→    }
  1591→    if (seedRound !== null && reportFeedsVolumeMainline(uid, state)) {
  1592→      return finish(Math.max(2, seedRound + 1));
  1593→    }
  1594→    return finish(Math.max(1, (particleSourceRound ?? maxSourceRound) || seedRound || 1));
  1595→  }
  1596→
  1597→  if (reportIsPickingNode(node)) {
  1598→    return finish(Math.max(1, (particleSourceRound ?? maxSourceRound) || 1));
  1599→  }
  1600→
  1601→  if (reportIsParticleAuxNode(node)) {
  1602→    return finish(particleSourceRound ?? maxSourceRound);
  1603→  }
  1604→
  1605→  if (/class_2D|select_2D|rebalance_classes_2D|class_probability_filter/i.test(type)) {
  1606→    return finish(Math.max(1, particleSourceRound ?? maxSourceRound));
  1607→  }
  1608→
  1609→  if (/homo_abinit|import_volumes|import_templates|create_templates|hetero|nonuniform|homo_refine|local_refine|class_3D|var_3D|align_3D|homo_reconstruct|sym_expand|particle_subtract|volume_tools|volume_alignment|local_resolution|sharpen|fsc3D|cryodrgn|relion|helix|auto3Dre/i.test(type)) {
  1610→    return finish(particleSourceRound ?? maxSourceRound);
  1611→  }
  1612→
  1613→  return finish(particleSourceRound ?? maxSourceRound);
  1614→}
