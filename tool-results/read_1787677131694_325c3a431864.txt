   123→function UploadJsonPanel({ loaded, onLoad }: Props) {
   124→  const inputRef = useRef<HTMLInputElement>(null);
   125→  const [dragOver, setDragOver] = useState(false);
   126→  const [parsing, setParsing] = useState(false);
   127→
   128→  const handleFile = useCallback(async (file: File) => {
   129→    setParsing(true);
   130→    try {
   131→      const text = await file.text();
   132→      const json = JSON.parse(text);
   133→      let jobs: unknown[] = [];
   134→      let projectUid = "P";
   135→      if (Array.isArray(json)) {
   136→        jobs = json;
   137→      } else if (json && typeof json === "object") {
   138→        const obj = json as Record<string, unknown>;
   139→        if (Array.isArray(obj.jobs)) jobs = obj.jobs;
   140→        if (typeof obj.project_uid === "string") projectUid = obj.project_uid;
   141→      }
   142→      if (jobs.length === 0) {
   143→        throw new Error("No jobs array found in JSON.");
   144→      }
   145→      onLoad({
   146→        raw: json,
   147→        projectUid,
   148→        jobCount: jobs.length,
   149→        source: "upload",
   150→      });
   151→      toast.success(`Loaded ${jobs.length} jobs from ${file.name}`);
   152→    } catch (err) {
   153→      const msg = err instanceof Error ? err.message : String(err);
   154→      toast.error(`Failed to parse JSON: ${msg}`);
   155→    } finally {
   156→      setParsing(false);
   157→    }
   158→  }, [onLoad]);
   159→
   160→  const onDrop = useCallback((e: React.DragEvent) => {
   161→    e.preventDefault();
   162→    setDragOver(false);
   163→    const file = e.dataTransfer.files?.[0];
   164→    if (file) handleFile(file);
   165→  }, [handleFile]);
   166→
   167→  return (
   168→    <div className="space-y-3">
   169→      <div
   170→        onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
   171→        onDragLeave={() => setDragOver(false)}
   172→        onDrop={onDrop}
   173→        onClick={() => inputRef.current?.click()}
   174→        className={`group flex cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed px-6 py-10 text-center transition-colors ${
   175→          dragOver ? "border-teal-400 bg-teal-50" : "border-slate-300 bg-slate-50/50 hover:border-teal-300 hover:bg-teal-50/40"
   176→        }`}
   177→      >
   178→        <input
   179→          ref={inputRef}
   180→          type="file"
   181→          accept=".json,application/json"
   182→          className="hidden"
   183→          onChange={(e) => {
   184→            const file = e.target.files?.[0];
   185→            if (file) handleFile(file);
   186→            e.target.value = "";
   187→          }}
   188→        />
   189→        <div className="mb-2 flex h-10 w-10 items-center justify-center rounded-full bg-teal-100 text-teal-600">
   190→          {parsing ? <Loader2 className="h-5 w-5 animate-spin" /> : <UploadCloud className="h-5 w-5" />}
   191→        </div>
   192→        <p className="text-[13px] font-medium text-slate-700">
   193→          {parsing ? "Parsing…" : "Drop the CryoSmart metadata JSON here, or click to browse"}
   194→        </p>
   195→        <p className="mt-1 text-[11px] text-slate-500">
   196→          Accepts a top-level array of jobs, or <code className="rounded bg-slate-200 px-1 py-0.5 text-[10px]">{"{ jobs: [...] }"}</code> wrapper
   197→        </p>
   198→      </div>
   199→      <div className="rounded-lg border border-amber-200 bg-amber-50/60 p-2.5 text-[11.5px] text-amber-800">
   200→        <div className="flex items-start gap-2">
   201→          <AlertCircle className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   202→          <div>
   203→            <span className="font-medium">How to get this JSON?</span> Use the original Chrome extension&apos;s
   204→            <span className="mx-1 font-mono rounded bg-amber-100 px-1">可选：导出当前 Project 全部 metadata</span>
   205→            button, OR fetch <code className="rounded bg-amber-100 px-1">/api/projects/&lt;pid&gt;/jobs</code> from CryoSmart directly. See the <a href="#help" className="underline">Help section</a> below.
   206→          </div>
   207→        </div>
   208→      </div>
   209→    </div>
   210→  );
   211→}

   213→function LiveConnectPanel({ loaded, onLoad }: Props) {
   214→  const existing = loadSession();
   215→  const [baseUrl, setBaseUrl] = useState(existing?.baseUrl || "http://192.168.4.3:8080");
   216→  const [cookie, setCookie] = useState(existing?.cookie || "");
   217→  const [auth, setAuth] = useState(existing?.auth || "");
   218→  const [projectId, setProjectId] = useState("P52");
   219→  const [testing, setTesting] = useState(false);
   220→  const [testResult, setTestResult] = useState<{ ok: boolean; msg: string } | null>(null);
   221→  const [loadingJobs, setLoadingJobs] = useState(false);
   222→
   223→  const testConnection = useCallback(async () => {
   224→    setTesting(true);
   225→    setTestResult(null);
   226→    try {
   227→      const session: CryoSmartSession = { baseUrl: baseUrl.replace(/\/+$/, ""), cookie: cookie || undefined, auth: auth || undefined };
   228→      saveSession(session);
   229→      const result = await fetchProjectJobs(session, projectId);
   230→      setTestResult({ ok: true, msg: `OK — found ${result.jobs.length} jobs via ${result.source}` });
   231→      toast.success(`Connected: ${result.jobs.length} jobs found`);
   232→    } catch (err) {
   233→      const msg = err instanceof Error ? err.message : String(err);
   234→      setTestResult({ ok: false, msg });
   235→      toast.error(`Connection failed: ${msg}`);
   236→    } finally {
   237→      setTesting(false);
   238→    }
   239→  }, [baseUrl, cookie, auth, projectId]);
   240→
   241→  const loadJobs = useCallback(async () => {
   242→    setLoadingJobs(true);
   243→    try {
   244→      const session: CryoSmartSession = { baseUrl: baseUrl.replace(/\/+$/, ""), cookie: cookie || undefined, auth: auth || undefined };
   245→      saveSession(session);
   246→      const result = await fetchProjectJobs(session, projectId);
   247→      const jobUids: string[] = [];
   248→      for (const j of result.jobs as Array<Record<string, unknown>>) {
   249→        if (typeof j.uid === "string") jobUids.push(j.uid);
   250→        else if (typeof j.uid_num === "number") jobUids.push(`J${j.uid_num}`);
   251→      }
   252→      jobUids.sort();
   253→      onLoad({
   254→        raw: { jobs: result.jobs },
   255→        projectUid: projectId,
   256→        jobCount: result.jobs.length,
   257→        source: "live",
   258→        session,
   259→        liveJobUids: jobUids,
   260→      });
   261→      toast.success(`Loaded ${result.jobs.length} jobs from ${session.baseUrl}`);
   262→    } catch (err) {
   263→      const msg = err instanceof Error ? err.message : String(err);
   264→      toast.error(`Load jobs failed: ${msg}`);
   265→    } finally {
   266→      setLoadingJobs(false);
   267→    }
   268→  }, [baseUrl, cookie, auth, projectId, onLoad]);
   269→
   270→  return (
   271→    <div className="space-y-3">
   272→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   273→        <div className="space-y-1.5 sm:col-span-2">
   274→          <Label htmlFor="cryosmart-base" className="text-[12px] text-slate-600">CryoSmart Base URL</Label>
   275→          <Input
   276→            id="cryosmart-base"
   277→            value={baseUrl}
   278→            onChange={(e) => setBaseUrl(e.target.value)}
   279→            placeholder="http://192.168.4.3:8080"
   280→            className="h-9 font-mono text-[13px]"
   281→          />
   282→        </div>
   283→        <div className="space-y-1.5">
   284→          <Label htmlFor="cryosmart-cookie" className="text-[12px] text-slate-600">Session Cookie <span className="text-slate-400">(optional)</span></Label>
   285→          <Input
   286→            id="cryosmart-cookie"
   287→            value={cookie}
   288→            onChange={(e) => setCookie(e.target.value)}
   289→            placeholder="session=eyJ...; csrftoken=abc"
   290→            className="h-9 font-mono text-[12px]"
   291→          />
   292→        </div>
   293→        <div className="space-y-1.5">
   294→          <Label htmlFor="cryosmart-auth" className="text-[12px] text-slate-600">Authorization <span className="text-slate-400">(optional)</span></Label>
   295→          <Input
   296→            id="cryosmart-auth"
   297→            value={auth}
   298→            onChange={(e) => setAuth(e.target.value)}
   299→            placeholder="Bearer eyJ..."
   300→            className="h-9 font-mono text-[12px]"
   301→          />
   302→        </div>
   303→        <div className="space-y-1.5">
   304→          <Label htmlFor="cryosmart-pid" className="text-[12px] text-slate-600">Project ID</Label>
   305→          <Input
   306→            id="cryosmart-pid"
   307→            value={projectId}
   308→            onChange={(e) => setProjectId(e.target.value)}
   309→            placeholder="P52"
   310→            className="h-9 font-mono text-[13px]"
   311→          />
   312→        </div>
   313→      </div>
   314→
   315→      <div className="flex flex-wrap gap-2">
   316→        <Button variant="outline" size="sm" onClick={testConnection} disabled={testing || !baseUrl} className="h-8 text-[12px]">
   317→          {testing ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <Plug className="mr-1.5 h-3.5 w-3.5" />}
   318→          Test connection
   319→        </Button>
   320→        <Button size="sm" onClick={loadJobs} disabled={loadingJobs || !baseUrl || !projectId} className="h-8 text-[12px] bg-teal-600 hover:bg-teal-700">
   321→          {loadingJobs ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <Server className="mr-1.5 h-3.5 w-3.5" />}
   322→          Load jobs
   323→        </Button>
   324→      </div>
   325→
   326→      {testResult && (
   327→        <div className={`flex items-start gap-2 rounded-md border p-2.5 text-[12px] ${testResult.ok ? "border-emerald-200 bg-emerald-50 text-emerald-800" : "border-rose-200 bg-rose-50 text-rose-800"}`}>
   328→          {testResult.ok ? <CheckCircle2 className="mt-0.5 h-3.5 w-3.5 shrink-0" /> : <AlertCircle className="mt-0.5 h-3.5 w-3.5 shrink-0" />}
   329→          <span className="font-mono">{testResult.msg}</span>
   330→        </div>
   331→      )}
   332→
   333→      {loaded?.source === "live" && loaded.liveJobUids && loaded.liveJobUids.length > 0 && (
   334→        <div className="rounded-md border border-slate-200 bg-slate-50 p-2.5">
   335→          <div className="mb-1.5 flex items-center justify-between text-[11px] text-slate-500">
   336→            <span>{loaded.liveJobUids.length} jobs discovered</span>
   337→            <span>pick one in the Configure step →</span>
   338→          </div>
   339→          <ScrollArea className="h-20 rounded border border-slate-200 bg-white p-1.5">
   340→            <div className="flex flex-wrap gap-1">
   341→              {loaded.liveJobUids.map((uid) => (
   342→                <Badge key={uid} variant="outline" className="font-mono text-[10px]">{uid}</Badge>
   343→              ))}
   344→            </div>
   345→          </ScrollArea>
   346→        </div>
   347→      )}
   348→
   349→      {/* How it works + tutorial */}
   350→      <LiveConnectTutorial baseUrl={baseUrl} cryosmartOrigin={baseUrl.replace(/\/+$/, "")} />
   351→    </div>
   352→  );
