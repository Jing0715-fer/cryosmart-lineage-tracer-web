   170→   170→function traceNormalizeJsonText(text) {
   171→   171→  return String(text || "")
   172→   172→    .replace(/[“”]/g, "\"")
   173→   173→    .replace(/[‘’]/g, "'");
   174→   174→}

   176→   176→function traceFindJsonEnd(text, start) {
   177→   177→  let depth = 0;
   178→   178→  let inString = false;
   179→   179→  let escaped = false;
   180→   180→  for (let i = start; i < text.length; i += 1) {
   181→   181→    const ch = text[i];
   182→   182→    if (inString) {
   183→   183→      if (escaped) {
   184→   184→        escaped = false;
   185→   185→      } else if (ch === "\\") {
   186→   186→        escaped = true;
   187→   187→      } else if (ch === "\"") {
   188→   188→        inString = false;
   189→   189→      }
   190→   190→      continue;
   191→   191→    }
   192→   192→    if (ch === "\"") {
   193→   193→      inString = true;
   194→   194→    } else if (ch === "{") {
   195→   195→      depth += 1;
   196→   196→    } else if (ch === "}") {
   197→   197→      depth -= 1;
   198→   198→      if (depth === 0) return i + 1;
   199→   199→    }
   200→   200→  }
   201→   201→  return -1;
   202→   202→}

   204→   204→function traceExtractMetadataObject(text) {
   205→   205→  const normalized = traceNormalizeJsonText(text);
   206→   206→  const first = normalized.indexOf("{");
   207→   207→  if (first < 0) throw new Error("Metadata 页面里没有找到 JSON 开头。");
   208→   208→  const end = traceFindJsonEnd(normalized, first);
   209→   209→  if (end < 0) throw new Error("Metadata JSON 没有完整结束。");
   210→   210→  const rawJson = normalized.slice(first, end);
   211→   211→  const job = JSON.parse(rawJson);
   212→   212→  if (!job.uid) {
   213→   213→    const uidMatch = normalized.match(/\bJ\d+\b/i);
   214→   214→    if (uidMatch) job.uid = uidMatch[0].toUpperCase();
   215→   215→  }
   216→   216→  if (!Number.isInteger(job.uid_num)) {
   217→   217→    job.uid_num = traceJobNum(job.uid);
   218→   218→  }
   219→   219→  return { job, rawJson };
