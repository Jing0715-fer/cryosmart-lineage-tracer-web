    72→def parse_corr(text):
    73→    if text is None:
    74→        return None
    75→    if isinstance(text, (int, float)):
    76→        value = float(text)
    77→        return value if -1.0 <= value <= 1.0 else None
    78→    text = str(text)
    79→    m = re.search(r"correlation\s*=?\s*([-+]?\d*\.?\d+)", text, re.I)
    80→    if m:
    81→        value = float(m.group(1))
    82→        return value if -1.0 <= value <= 1.0 else None
    83→    return None

    85→def run_capture(session, cmd, command_log=None):
    86→    if command_log is not None:
    87→        command_log.append(cmd)
    88→    with tempfile.NamedTemporaryFile(mode="w+", suffix=".txt", delete=False) as tmp:
    89→        log_path = tmp.name
    90→    try:
    91→        cxrun(session, "log clear")
    92→        result = cxrun(session, cmd)
    93→        cxrun(session, "log save " + q(log_path))
    94→        with open(log_path, "r", encoding="utf-8", errors="ignore") as f:
    95→            log_text = f.read()
    96→        return result, log_text
    97→    finally:
    98→        if os.path.exists(log_path):
    99→            os.unlink(log_path)

   101→def measure_corr(session, vol_id, ref_id, command_log=None):
   102→    cmd = f"measure correlation #{vol_id} inMap #{ref_id}"
   103→    result, log_text = run_capture(session, cmd, command_log)
   104→    corr = parse_corr(result)
   105→    if corr is None:
   106→        corr = parse_corr(log_text)
   107→    return corr, str(result), log_text
   108→
   109→def fit_corr(session, vol_id, ref_id, search=None, command_log=None):
   110→    cmd = f"fit #{vol_id} in #{ref_id}"
   111→    if search is not None:
   112→        cmd += f" search {search}"
   113→    result, log_text = run_capture(session, cmd, command_log)
   114→    corr = parse_corr(result)
   115→    if corr is None:
   116→        corr = parse_corr(log_text)
   117→    if corr is None:
   118→        corr, _, _ = measure_corr(session, vol_id, ref_id, command_log)
   119→    return corr, str(result), log_text

   121→def open_volume(session, path, command_log=None):
   122→    cmd = "open " + q(path)
   123→    if command_log is not None:
   124→        command_log.append(cmd)
   125→    before = {m for m in session.models.list()}
   126→    cxrun(session, cmd)
   127→    after = [m for m in session.models.list() if m not in before]
   128→    volumes = [m for m in after if m.__class__.__name__ == "Volume"]
   129→    if not volumes:
   130→        raise RuntimeError("Opened file is not a Volume: " + str(path))
   131→    return volumes[-1]

   197→def mrc_candidates_from_summary(folder, summary):
   198→    candidates = {}
   199→    project = summary.get("project_uid") or ""
   200→    for node in summary.get("nodes") or []:
   201→        uid = node.get("uid")
   202→        if not uid:
   203→            continue
   204→        for item in normal_map_assets(node):
   205→            group = item.get("group") or "volume"
   206→            suffix = f"{group}.{item.get('result_name') or 'map'}"
   207→            path = folder / "maps" / safe_name(uid) / f"BJ.{safe_name(project)}.{safe_name(uid)}.{safe_name(suffix)}.mrc"
   208→            key = map_key_for_node_item(node, item)
   209→            candidates[key] = {
   210→                "key": key,
   211→                "uid": uid,
   212→                "job_type": node.get("job_type") or "",
   213→                "group": group,
   214→                "path": path,
   215→                "is_final": uid == summary.get("start_uid"),
   216→            }
   217→
   218→    start_uid = summary.get("start_uid")
   219→    for name in (summary.get("map_download_urls") or {}).keys():
   220→        path = folder / "maps" / safe_name(start_uid) / f"BJ.{safe_name(project)}.{safe_name(start_uid)}.{safe_name(name)}.mrc"
   221→        key = canonical_map_key(start_uid, name)
   222→        candidates.setdefault(key, {
   223→            "key": key,
   224→            "uid": start_uid,
   225→            "job_type": (summary.get("start_job") or {}).get("job_type") or "",
   226→            "group": name,
   227→            "path": path,
   228→            "is_final": True,
   229→        })
   230→
   231→    maps_dir = folder / "maps"
   232→    for path in sorted(maps_dir.glob("**/*")) if maps_dir.exists() else []:
   233→        if path.is_file() and path.suffix.lower() in (".mrc", ".map"):
   234→            m = re.search(r"\.(J\d+)\.", "." + path.name)
   235→            uid = m.group(1) if m else path.parent.name
   236→            group = group_from_mrc_name(path)
   237→            key = canonical_map_key(uid, group)
   238→            candidates.setdefault(key, {
   239→                "key": key,
   240→                "uid": uid,
   241→                "job_type": "",
   242→                "group": group,
   243→                "path": path,
   244→                "is_final": uid == start_uid,
   245→            })
   246→
   247→    unique = []
   248→    seen_paths = set()
   249→    for item in candidates.values():
   250→        if not item["path"].exists():
   251→            continue
   252→        resolved = item["path"].resolve()
   253→        if resolved in seen_paths:
   254→            continue
   255→        seen_paths.add(resolved)
   256→        unique.append(item)
   257→    return unique

   269→def class_index_from_group(group):

   273→def target_map_groups_for_edge(edge, target_node, item_by_key):
   274→    target_uid = edge.get("target")
   275→    source_group = map_group_key(edge.get("source_group"))
   276→    source_class = class_index_from_group(source_group)
   277→    target_groups = []
   278→    if target_node:
   279→        for map_item in normal_map_assets(target_node):
   280→            group = map_group_key(map_item.get("group") or "volume")
   281→            if group not in target_groups:
   282→                target_groups.append(group)
   283→
   284→    candidates = []
   285→    if canonical_map_key(target_uid, source_group) in item_by_key:
   286→        candidates.append(source_group)
   287→    if source_class is not None:
   288→        for group in target_groups:
   289→            if class_index_from_group(group) == source_class and group not in candidates:
   290→                candidates.append(group)
   291→    if not candidates and "volume" in target_groups:
   292→        candidates.append("volume")
   293→    if not candidates and len(target_groups) == 1:
   294→        candidates.append(target_groups[0])
   295→    return [group for group in candidates if canonical_map_key(target_uid, group) in item_by_key]

   297→def build_map_lineage(summary, items):
   298→    item_by_key = {item["key"]: item for item in items}
   299→    nodes_by_uid = {node.get("uid"): node for node in summary.get("nodes") or [] if node.get("uid")}
   300→    edges = []
   301→    seen = set()
   302→    warnings = []
   303→
   304→    for edge in summary.get("edges") or []:
   305→        if edge.get("input_type") != "volume":
   306→            continue
   307→        source_uid = edge.get("source")
   308→        target_uid = edge.get("target")
   309→        source_group = map_group_key(edge.get("source_group"))
   310→        source_key = canonical_map_key(source_uid, source_group)
   311→        if source_key not in item_by_key:
   312→            warnings.append(f"skip volume edge {source_uid}.{source_group} -> {target_uid}: source MRC missing")
   313→            continue
   314→        target_node = nodes_by_uid.get(target_uid)
   315→        for target_group in target_map_groups_for_edge(edge, target_node, item_by_key):
   316→            target_key = canonical_map_key(target_uid, target_group)
   317→            edge_key = (source_key, target_key)
   318→            if edge_key in seen:
   319→                continue
   320→            seen.add(edge_key)
   321→            edges.append({
   322→                "source": source_key,
   323→                "target": target_key,
   324→                "reason": f"{source_uid}.{source_group} -> {target_uid}.{target_group}",
   325→            })
   326→
   327→    incoming = {}
   328→    outgoing = {}
   329→    for edge in edges:
   330→        outgoing.setdefault(edge["source"], []).append(edge["target"])
   331→        incoming.setdefault(edge["target"], []).append(edge["source"])
   332→
   333→    start_keys = [key for key in item_by_key if key not in incoming]
   334→    if not start_keys:
   335→        start_keys = sorted(item_by_key)
   336→        warnings.append("No start map inferred; treating every downloaded map as a possible start.")
   337→
   338→    chains = []
   339→
   340→    def walk(start_key, current_key, path):
   341→        children = sorted(outgoing.get(current_key) or [])
   342→        if not children:
   343→            chains.append({"start": start_key, "terminal": current_key, "keys": list(path)})
   344→            return
   345→        for child in children:
   346→            if child in path:
   347→                warnings.append(f"Cycle detected in map lineage: {' -> '.join(path + [child])}")
   348→                continue
   349→            walk(start_key, child, path + [child])
   350→
   351→    for start_key in sorted(start_keys, key=lambda key: (parse_job_num(key), key)):
   352→        walk(start_key, start_key, [start_key])
   353→    if not chains:
   354→        chains = [{"start": key, "terminal": key, "keys": [key]} for key in sorted(item_by_key)]
   355→
   356→    return {
   357→        "items": item_by_key,
   358→        "edges": edges,
   359→        "incoming": incoming,
   360→        "outgoing": outgoing,
   361→        "starts": sorted(start_keys, key=lambda key: (parse_job_num(key), key)),
   362→        "chains": chains,
   363→        "warnings": warnings,
   364→    }

   366→def chain_label(chain):
   367→    return " -> ".join(chain.get("keys") or [])

   369→def chain_for_key(lineage, key):
   370→    containing = [chain for chain in lineage.get("chains") or [] if key in (chain.get("keys") or [])]
   371→    if not containing:
   372→        return None
   373→    return sorted(containing, key=lambda chain: (-len(chain.get("keys") or []), chain_label(chain)))[0]

   390→def test_candidate(session, item, reference, ref_vol, ref_density, spec):
   391→    commands = []
   392→    errors = []
   393→    opened = []
   394→    active = None
   395→    try:
   396→        vol = open_volume(session, item["path"], commands)
   397→        opened.append(vol)
   398→        vol.name = f"{item['key']}__{spec['name']}"
   399→        set_volume_display(session, vol, ref_density, commands)
   400→        active = vol
   401→
   402→        if spec["kind"] == "rotate":
   403→            cmd = f"turn {spec['axis']} {spec['angle']} models #{vol.id_string} center #{vol.id_string}"
   404→            commands.append(cmd)
   405→            cxrun(session, cmd)
   406→            commands.append("wait 1")
   407→            cxrun(session, "wait 1")
   408→        elif spec["kind"] == "flip":
   409→            flip_id = max([m.id[0] for m in session.models.list() if getattr(m, "id", None)] + [0]) + 1
   410→            cmd = f"volume flip #{vol.id_string} axis {spec['axis']} modelId {flip_id}"
   411→            commands.append(cmd)
   412→            cxrun(session, cmd)
   413→            flip_model = next((m for m in volume_models(session) if m.id[0] == flip_id), None)
   414→            if flip_model is None:
   415→                raise RuntimeError("flip model was not created")
   416→            flip_model.name = f"{item['key']}__{spec['name']}"
   417→            set_volume_display(session, flip_model, ref_density, commands)
   418→            active = flip_model
   419→            close_model(session, vol, commands)
   420→            opened = [flip_model]
   421→
   422→        pre_corr, pre_raw, pre_log = measure_corr(session, active.id_string, ref_vol.id_string, commands)
   423→        fit1_corr, fit1_raw, fit1_log = fit_corr(session, active.id_string, ref_vol.id_string, search=None, command_log=commands)
   424→        fit2_corr, fit2_raw, fit2_log = fit_corr(session, active.id_string, ref_vol.id_string, search=FIT_SEARCH, command_log=commands)
   425→
   426→        score = fit2_corr if fit2_corr is not None else fit1_corr
   427→        if score is None:
   428→            score = pre_corr
   429→
   430→        return {
   431→            "candidate": spec["name"],
   432→            "kind": spec["kind"],
   433→            "model": active,
   434→            "model_id": active.id_string,
   435→            "score": score,
   436→            "pre_corr": pre_corr,
   437→            "fit_no_search_corr": fit1_corr,
   438→            "fit_search_corr": fit2_corr,
   439→            "position": position_text(active),
   440→            "commands": commands,
   441→            "errors": errors,
   442→            "raw": {
   443→                "pre": pre_raw,
   444→                "fit_no_search": fit1_raw,
   445→                "fit_search": fit2_raw,
   446→                "pre_log": pre_log,
   447→                "fit_no_search_log": fit1_log,
   448→                "fit_search_log": fit2_log,
   449→            },
   450→            "opened_models": opened,
   451→        }
   452→    except Exception as exc:
   453→        errors.append(str(exc))
   454→        for model in opened:
   455→            close_model(session, model, commands)
   456→        return {
   457→            "candidate": spec["name"],
   458→            "kind": spec["kind"],
   459→            "model": None,
   460→            "model_id": "",
   461→            "score": None,
   462→            "pre_corr": None,
   463→            "fit_no_search_corr": None,
   464→            "fit_search_corr": None,
   465→            "position": "",
   466→            "commands": commands,
   467→            "errors": errors,
   468→            "raw": {},
   469→            "opened_models": [],
   470→        }

   522→            lines.append(current)
   523→            lines.append("=" * 80)
   524→        mark = "CHOSEN" if row.get("chosen") else "test"
   525→        lines.append(

   542→# ==================== 视角自动优化函数（来自第三个脚本） ====================
   543→def score_current_view(session):
   544→    """截取当前视角缩略图，计算非白色像素比例作为评分（地图覆盖率）"""
   545→    if np is None or Image is None:
   546→        return 0.0
   547→    with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
   548→        path = tmp.name
   549→    try:
   550→        cxrun(session, f"save {q(path)} width {SCORE_THUMB_SIZE} height {SCORE_THUMB_SIZE} supersample 1 transparentBackground false")
   551→        img = Image.open(path).convert("RGB")
   552→        arr = np.array(img)
   553→        bg = np.all(arr > 240, axis=-1)
   554→        score = float((arr.shape[0] * arr.shape[1] - np.sum(bg)) / (arr.shape[0] * arr.shape[1]))
   555→        return score
   556→    except Exception as e:
   557→        session.logger.warning(f"视角评分失败: {e}")
   558→        return 0.0
   559→    finally:
   560→        if os.path.exists(path):
   561→            os.unlink(path)

   563→def select_best_90_view(session, test_vol):
   564→    """
   565→    基于当前视角（需先手动调好基准），测试5个方向（基准、Y±90、X±90），
   566→    选择地图占画面比例最大的方向，并应用该视角。
   567→    """
   568→    session.logger.info("===== 90° 视角自动优化 =====")
   569→
   570→    # 保存当前视角为基准
   571→    cxrun(session, "view name base_view")
   572→    cxrun(session, "wait 1")
   573→
   574→    # 只显示测试地图
   575→    all_vols = volume_models(session)
   576→    for vol in all_vols:
   577→        if vol.id_string == test_vol.id_string:
   578→            cxrun(session, f"show #{vol.id_string} models")
   579→        else:
   580→            cxrun(session, f"hide #{vol.id_string} models")
   581→    cxrun(session, "wait 1")
   582→
   583→    candidates = [
   584→        ("base", 0, 0),
   585→        ("y+90", 0, 90),
   586→        ("y-90", 0, -90),
   587→        ("x+90", 90, 0),
   588→        ("x-90", -90, 0),
   589→    ]
   590→
   591→    best_score = -1.0
   592→    best_desc = "base"
   593→    best_rot = (0, 0)
   594→
   595→    for desc, x_rot, y_rot in candidates:
   596→        cxrun(session, "view base_view")
   597→        cxrun(session, "wait 1")
   598→        if x_rot != 0:
   599→            cxrun(session, f"turn x {x_rot}")
   600→        if y_rot != 0:
   601→            cxrun(session, f"turn y {y_rot}")
   602→        cxrun(session, "wait 1")
   603→        score = score_current_view(session)
   604→        session.logger.info(f"  {desc}: X={x_rot}°, Y={y_rot}°, Score={score:.4f}")
   605→        if score > best_score:
   606→            best_score = score
   607→            best_desc = desc
   608→            best_rot = (x_rot, y_rot)
   609→
   610→    # 应用最佳视角
   611→    session.logger.info(f"最佳视角: {best_desc} (X={best_rot[0]}°, Y={best_rot[1]}°), Score={best_score:.4f}")
   612→    cxrun(session, "view base_view")
   613→    cxrun(session, "wait 1")
   614→    if best_rot[0] != 0:
   615→        cxrun(session, f"turn x {best_rot[0]}")
   616→    if best_rot[1] != 0:
   617→        cxrun(session, f"turn y {best_rot[1]}")
   618→    cxrun(session, "wait 1")
   619→
   620→    # 恢复显示所有地图
   621→    for vol in all_vols:
   622→        cxrun(session, f"show #{vol.id_string} models")
   623→    cxrun(session, "wait 1")
   624→
   625→    return {"desc": best_desc, "x": best_rot[0], "y": best_rot[1], "score": best_score}

   628→# ==================== 主流程 ====================
   629→def main(session, lineage_folder=None):
   630→    if lineage_folder is None:
   631→        argv = list(sys.argv[1:])
   632→        if argv and Path(argv[0]).expanduser().exists():
   633→            lineage_folder = argv[0]
   634→        else:
   635→            script_path = Path(__file__).expanduser().resolve() if "__file__" in globals() else None
   636→            lineage_folder = script_path.parent if script_path else os.getcwd()
   637→    folder = Path(lineage_folder).expanduser().resolve()
   638→    if folder.is_file():
   639→        folder = folder.parent
   640→
   641→    session.logger.info("===== CryoSmart 自动对齐 + 视角优化（无导出） =====")
   642→    session.logger.info(f"工作目录: {folder}")
   643→
   644→    # ---- 1. 加载谱系数据 ----
   645→    summary_path = find_lineage_json(folder)
   646→    summary = json.loads(summary_path.read_text(encoding="utf-8"))
   647→    items = mrc_candidates_from_summary(folder, summary)
   648→    if not items:
   649→        raise RuntimeError("没有找到可用的 MRC 文件")
   650→    reference = choose_reference(items, summary)
   651→    if not reference:
   652→        raise RuntimeError("无法确定参考地图")
   653→
   654→    session.logger.info(f"参考地图: {reference['key']} -> {reference['path']}")
   655→
   656→    # ---- 2. 构建 map 谱系 ----
   657→    lineage = build_map_lineage(summary, items)
   658→    manifest = {
   659→        "lineage_folder": str(folder),
   660→        "lineage_json": str(summary_path),
   661→        "reference_key": reference["key"],
   662→        "reference_path": str(reference["path"]),
   663→        "fit_search": FIT_SEARCH,
   664→        "map_lineage": {
   665→            "starts": lineage["starts"],
   666→            "edges": lineage["edges"],
   667→            "chains": lineage["chains"],
   668→            "warnings": lineage["warnings"],
   669→        },
   670→        "models": {},
   671→        "terminal_results": {},
   672→        "warnings": list(lineage["warnings"]),
   673→    }
   674→
   675→    # ---- 3. 打开参考地图 ----
   676→    ref_commands = []
   677→    ref_vol = open_volume(session, reference["path"], ref_commands)
   678→    ref_vol.name = reference["key"]
   679→    ref_density = max_density(ref_vol)
   680→    set_volume_display(session, ref_vol, ref_density, ref_commands)
   681→    manifest["models"][reference["key"]] = {
   682→        "role": "reference",
   683→        "model_id": ref_vol.id_string,
   684→        "source_mrc": str(reference["path"]),
   685→        "position": position_text(ref_vol),
   686→        "commands": ref_commands,
   687→    }
   688→
   689→    loaded_by_key = {reference["key"]: {"model": ref_vol, **reference, "score": 1.0}}
   690→    terminal_results = {reference["key"]: loaded_by_key[reference["key"]]}
   691→    all_candidate_rows = []
   692→
   693→    # ---- 4. 对每个终端地图测试候选 ----
   694→    terminal_keys = sorted({chain["terminal"] for chain in lineage["chains"]})
   695→    for terminal_key in terminal_keys:
   696→        if terminal_key == reference["key"]:
   697→            continue
   698→        item = lineage["items"].get(terminal_key)
   699→        if not item:
   700→            continue
   701→        session.logger.info(f"测试终端地图: {terminal_key}")
   702→
   703→        candidate_rows = []
   704→        for spec in candidate_specs():
   705→            row = test_candidate(session, item, reference, ref_vol, ref_density, spec)
   706→            candidate_rows.append(row)
   707→            session.logger.info(
   708→                f"  {spec['name']}: score={row.get('score')} "
   709→                f"fit1={row.get('fit_no_search_corr')} fit_search={row.get('fit_search_corr')}"
   710→            )
   711→
   712→        best = choose_best(candidate_rows)
   713→        if GENERATE_CXC:
   714→            cxc_dir = folder / "debug_cxc"
   715→            cxc_dir.mkdir(exist_ok=True)
   716→            cxc_path = write_candidate_cxc(cxc_dir, item, reference, candidate_rows)
   717→        else:
   718→            cxc_path = None
   719→
   720→        for row in candidate_rows:
   721→            row["map_key"] = item["key"]
   722→            row["chosen"] = bool(best and row is best)
   723→            row["reference_key"] = reference["key"]
   724→            row["source_mrc"] = str(item["path"])
   725→            row["reference_mrc"] = str(reference["path"])
   726→            row["cxc_file"] = str(cxc_path) if cxc_path else ""
   727→            row["errors"] = "; ".join(row.get("errors") or [])
   728→            all_candidate_rows.append(row)
   729→
   730→        if best is None:
   731→            manifest["warnings"].append(f"没有找到有效候选: {item['key']}")
   732→            continue
   733→
   734→        for row in candidate_rows:
   735→            if row is best:
   736→                continue
   737→            for model in row.get("opened_models") or []:
   738→                close_model(session, model)
   739→
   740→        best_model = best["model"]
   741→        best_model.name = item["key"] + "__BEST_" + best["candidate"]
   742→        result = {
   743→            **item,
   744→            "model": best_model,
   745→            "score": best["score"],
   746→            "candidate": best["candidate"],
   747→            "model_id": best_model.id_string,
   748→            "position": position_text(best_model),
   749→            "cxc_file": str(cxc_path) if cxc_path else "",
   750→        }
   751→        terminal_results[terminal_key] = result
   752→        loaded_by_key[terminal_key] = result
   753→        manifest["terminal_results"][terminal_key] = {
   754→            "chosen_candidate": best["candidate"],
   755→            "score": best["score"],
   756→            "model_id": best_model.id_string,
   757→            "source_mrc": str(item["path"]),
   758→            "position": position_text(best_model),
   759→        }
   760→        manifest["models"][item["key"]] = {
   761→            "role": "chain_terminal",
   762→            "model_id": best_model.id_string,
   763→            "chosen_candidate": best["candidate"],
   764→            "score": best["score"],
   765→            "source_mrc": str(item["path"]),
   766→            "position": position_text(best_model),
   767→        }
   768→        session.logger.info(f"  选中 {terminal_key}: {best['candidate']} score={best['score']}")
   769→
   770→    # ---- 5. 继承变换给链上非终端成员 ----
   771→    for item in items:
   772→        if item["key"] in loaded_by_key:
   773→            continue
   774→        chain, terminal_result = lineage_terminal_results_for_key(lineage, terminal_results, item["key"])
   775→        if not terminal_result or not terminal_result.get("model"):
   776→            manifest["warnings"].append(f"无法继承变换: {item['key']}（缺少终端结果）")
   777→            continue
   778→        commands = []
   779→        try:
   780→            vol = open_volume(session, item["path"], commands)
   781→            vol.name = item["key"] + "__inherited"
   782→            set_volume_display(session, vol, ref_density, commands)
   783→            vol.position = terminal_result["model"].position
   784→            loaded_by_key[item["key"]] = {**item, "model": vol}
   785→            manifest["models"][item["key"]] = {
   786→                "role": "chain_member_inherited",
   787→                "model_id": vol.id_string,
   788→                "source_mrc": str(item["path"]),
   789→                "inherited_from_terminal": chain["terminal"] if chain else "",
   790→                "terminal_candidate": terminal_result.get("candidate", ""),
   791→                "terminal_score": terminal_result.get("score", ""),
   792→                "position": position_text(vol),
   793→                "commands": commands,
   794→            }
   795→        except Exception as exc:
   796→            manifest["warnings"].append(f"继承失败 {item['key']}: {exc}")

   802→        write_debug_logs(out_dir, manifest, all_candidate_rows)

   805→    for vol in volume_models(session):
   806→        cxrun(session, f"show #{vol.id_string} models")
   807→    cxrun(session, "set bgColor white")
   808→    cxrun(session, "view pad 0.25")
   809→    cxrun(session, "wait 1")
   810→
   811→    session.logger.info("\n" + "=" * 60)
   812→    session.logger.info("请手动旋转/缩放视角到您想要的基准方向（使地图居中且特征明显）。")
   813→    session.logger.info("调整好后，脚本将自动测试 5 个方向（基准、Y±90、X±90）并选出最优视角。")

   815→    session.logger.info("=" * 60 + "\n")
   816→

   824→    manifest_path.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")

   827→    session.logger.info("===== 对齐与视角优化完成 =====")
   828→    session.logger.info(f"最佳视角: {best_view['desc']} (X={best_view['x']}°, Y={best_view['y']}°)")
