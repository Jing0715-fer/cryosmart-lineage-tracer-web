   305→   305→def build_map_lineage(summary, items):
   306→   306→    item_by_key = {item["key"]: item for item in items}
   307→   307→    nodes_by_uid = {node.get("uid"): node for node in summary.get("nodes") or [] if node.get("uid")}
   308→   308→    edges = []
   309→   309→    seen = set()
   310→   310→    warnings = []
   311→   311→
   312→   312→    for edge in summary.get("edges") or []:
   313→   313→        if edge.get("input_type") != "volume":
   314→   314→            continue
   315→   315→        source_uid = edge.get("source")
   316→   316→        target_uid = edge.get("target")
   317→   317→        source_group = map_group_key(edge.get("source_group"))
   318→   318→        source_key = canonical_map_key(source_uid, source_group)
   319→   319→        if source_key not in item_by_key:
   320→   320→            warnings.append(f"skip volume edge {source_uid}.{source_group} -> {target_uid}: source MRC missing")
   321→   321→            continue
   322→   322→        target_node = nodes_by_uid.get(target_uid)
   323→   323→        for target_group in target_map_groups_for_edge(edge, target_node, item_by_key):
   324→   324→            target_key = canonical_map_key(target_uid, target_group)
   325→   325→            edge_key = (source_key, target_key)
   326→   326→            if edge_key in seen:
   327→   327→                continue
   328→   328→            seen.add(edge_key)
   329→   329→            edges.append({
   330→   330→                "source": source_key,
   331→   331→                "target": target_key,
   332→   332→                "reason": f"{source_uid}.{source_group} -> {target_uid}.{target_group}",
   333→   333→            })
   334→   334→
   335→   335→    incoming = {}
   336→   336→    outgoing = {}
   337→   337→    for edge in edges:
   338→   338→        outgoing.setdefault(edge["source"], []).append(edge["target"])
   339→   339→        incoming.setdefault(edge["target"], []).append(edge["source"])
   340→   340→
   341→   341→    start_keys = [key for key in item_by_key if key not in incoming]
   342→   342→    if not start_keys:
   343→   343→        start_keys = sorted(item_by_key)
   344→   344→        warnings.append("No start map inferred; treating every downloaded map as a possible start.")
   345→   345→
   346→   346→    chains = []
   347→   347→
   348→   348→    def walk(start_key, current_key, path):
   349→   349→        children = sorted(outgoing.get(current_key) or [])
   350→   350→        if not children:
   351→   351→            chains.append({"start": start_key, "terminal": current_key, "keys": list(path)})
   352→   352→            return
   353→   353→        for child in children:
   354→   354→            if child in path:
   355→   355→                warnings.append(f"Cycle detected in map lineage: {' -> '.join(path + [child])}")
   356→   356→                continue
   357→   357→            walk(start_key, child, path + [child])
   358→   358→
   359→   359→    for start_key in sorted(start_keys, key=lambda key: (parse_job_num(key), key)):
   360→   360→        walk(start_key, start_key, [start_key])
   361→   361→    if not chains:
   362→   362→        chains = [{"start": key, "terminal": key, "keys": [key]} for key in sorted(item_by_key)]
   363→   363→
   364→   364→    return {
   365→   365→        "items": item_by_key,
   366→   366→        "edges": edges,
   367→   367→        "incoming": incoming,
   368→   368→        "outgoing": outgoing,
   369→   369→        "starts": sorted(start_keys, key=lambda key: (parse_job_num(key), key)),
   370→   370→        "chains": chains,
   371→   371→        "warnings": warnings,
   372→   372→    }

   622→   622→def content_bbox(image):
   623→   623→    arr = np.asarray(image.convert("RGBA"))
   624→   624→    alpha = arr[:, :, 3] > 10
   625→   625→    distance_from_white = np.max(255 - arr[:, :, :3].astype(np.int16), axis=2)
   626→   626→    nonwhite = alpha & (distance_from_white > WHITE_TRIM_DELTA)
   627→   627→    if not nonwhite.any():
   628→   628→        return None
   629→   629→    ys, xs = np.where(nonwhite)
   630→   630→    return int(xs.min()), int(ys.min()), int(xs.max()) + 1, int(ys.max()) + 1

   632→   632→def expand_box_to_aspect(box, size, target_aspect):
   633→   633→    left, top, right, bottom = [float(v) for v in box]
   634→   634→    w, h = size
   635→   635→    box_w = right - left
   636→   636→    box_h = bottom - top
   637→   637→    if box_w <= 0 or box_h <= 0:
   638→   638→        return 0, 0, w, h
   639→   639→    current = box_w / box_h
   640→   640→    if abs(current - target_aspect) < 0.002:
   641→   641→        return int(round(left)), int(round(top)), int(round(right)), int(round(bottom))
   642→   642→    if current > target_aspect:
   643→   643→        needed_h = box_w / target_aspect
   644→   644→        extra = needed_h - box_h
   645→   645→        top -= extra / 2
   646→   646→        bottom += extra / 2
   647→   647→        if top < 0:
   648→   648→            bottom -= top
   649→   649→            top = 0
   650→   650→        if bottom > h:
   651→   651→            top -= bottom - h
   652→   652→            bottom = h
   653→   653→        top = max(0, top)
   654→   654→    else:
   655→   655→        needed_w = box_h * target_aspect
   656→   656→        extra = needed_w - box_w
   657→   657→        left -= extra / 2
   658→   658→        right += extra / 2
   659→   659→        if left < 0:
   660→   660→            right -= left
   661→   661→            left = 0
   662→   662→        if right > w:
   663→   663→            left -= right - w
   664→   664→            right = w
   665→   665→        left = max(0, left)
   666→   666→    return int(round(max(0, left))), int(round(max(0, top))), int(round(min(w, right))), int(round(min(h, bottom)))

   668→   668→def common_trim_box(entries):
   669→   669→    margins = []
   670→   670→    image_size = None
   671→   671→    for _, data in entries:
   672→   672→        src = Path(data.get("png") or "")
   673→   673→        if not src.exists():
   674→   674→            continue
   675→   675→        image = Image.open(src).convert("RGBA")
   676→   676→        image_size = image.size
   677→   677→        box = content_bbox(image)
   678→   678→        if box:
   679→   679→            left, top, right, bottom = box
   680→   680→            w, h = image.size
   681→   681→            margins.append((left, top, w - right, h - bottom))
   682→   682→    if not margins or not image_size:
   683→   683→        return None
   684→   684→    w, h = image_size
   685→   685→    min_left = min(item[0] for item in margins)
   686→   686→    min_top = min(item[1] for item in margins)
   687→   687→    min_right = min(item[2] for item in margins)
   688→   688→    min_bottom = min(item[3] for item in margins)
   689→   689→    keep_x = max(MIN_TRIM_PADDING_PX, int(round(w * EXTRA_WHITE_MARGIN_FRACTION)))
   690→   690→    keep_y = max(MIN_TRIM_PADDING_PX, int(round(h * EXTRA_WHITE_MARGIN_FRACTION)))
   691→   691→    crop_left = max(0, min_left - keep_x)
   692→   692→    crop_top = max(0, min_top - keep_y)
   693→   693→    crop_right = max(0, min_right - keep_x)
   694→   694→    crop_bottom = max(0, min_bottom - keep_y)
   695→   695→    box = (crop_left, crop_top, w - crop_right, h - crop_bottom)
   696→   696→    return expand_box_to_aspect(box, image_size, w / h)

   795→   795→def score_current_view(session):
   796→   796→    """截取当前视角缩略图，计算非白色像素比例作为评分（地图覆盖率）"""
   797→   797→    if np is None or Image is None:
   798→   798→        return 0.0
   799→   799→    with tempfile.NamedTemporaryFile(suffix=".png", delete=False) as tmp:
   800→   800→        path = tmp.name
   801→   801→    try:
   802→   802→        cxrun(session, f"save {q(path)} width {SCORE_THUMB_SIZE} height {SCORE_THUMB_SIZE} supersample 1 transparentBackground false")
   803→   803→        img = Image.open(path).convert("RGB")
   804→   804→        arr = np.array(img)
   805→   805→        bg = np.all(arr > 240, axis=-1)
   806→   806→        score = float((arr.shape[0] * arr.shape[1] - np.sum(bg)) / (arr.shape[0] * arr.shape[1]))
   807→   807→        return score
   808→   808→    except Exception as e:
   809→   809→        session.logger.warning(f"视角评分失败: {e}")
   810→   810→        return 0.0
   811→   811→    finally:
   812→   812→        if os.path.exists(path):
   813→   813→            os.unlink(path)

   815→   815→def select_best_90_view(session, test_vol):
   816→   816→    """
   817→   817→    基于当前视角（需先手动调好基准），测试5个方向（基准、Y±90、X±90），
   818→   818→    选择地图占画面比例最大的方向，并应用该视角。
   819→   819→    """
   820→   820→    session.logger.info("===== 90° 视角自动优化 =====")
   821→   821→

   823→   823→    cxrun(session, "view name base_view")
   824→   824→    cxrun(session, "wait 1")
   825→   825→
   826→   826→    # 只显示测试地图
   827→   827→    all_vols = volume_models(session)
   828→   828→    for vol in all_vols:
   829→   829→        if vol.id_string == test_vol.id_string:
   830→   830→            cxrun(session, f"show #{vol.id_string} models")
   831→   831→        else:
   832→   832→            cxrun(session, f"hide #{vol.id_string} models")
   833→   833→    cxrun(session, "wait 1")
   834→   834→
   835→   835→    candidates = [
   836→   836→        ("base", 0, 0),
   837→   837→        ("y+90", 0, 90),
   838→   838→        ("y-90", 0, -90),
   839→   839→        ("x+90", 90, 0),
   840→   840→        ("x-90", -90, 0),
